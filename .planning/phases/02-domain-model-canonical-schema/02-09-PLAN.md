    ```go
    package ingest

    // Event is Shifter's narrow view of the ChirpStack v4 uplink event JSON.
    // We extract only the fields the ingest pipeline needs; everything else
    // (txInfo, deduplicationId etc.) goes through unread but is preserved in
    // the raw_payload column so per-meter detail (Phase 4) can show the firehose.
    type Event struct {
        DevEUI         string         // lowercase 16-char hex
        ApplicationID  string
        FCnt           uint32
        FPort          uint8
        DeduplicationID string
        Data           []byte         // base64-decoded LoRaWAN frame payload
        DecodedObject  map[string]any // QuickJS codec output — arbitrary shape
        RxInfo         []RxInfo
        DeviceTime     *time.Time     // sometimes set by device — diagnostic
        GatewayRxTime  *time.Time     // earliest rxInfo[].time — diagnostic
    }

    type RxInfo struct {
        GatewayID string
        RSSI      int16
        SNR       float32
        Time      *time.Time
    }

    // DecodeChirpStackEvent parses the JSON payload of an ChirpStack MQTT
    // uplink event. Returns the strongly-typed Event. Errors only on JSON
    // parse failure — missing/optional fields → zero values; the pipeline's
    // quality flag handles missing canonical fields downstream.
    func DecodeChirpStackEvent(payload []byte) (Event, error) {
        var raw struct {
            DeviceInfo struct {
                DevEui string `json:"devEui"`
                ApplicationId string `json:"applicationId"`
            } `json:"deviceInfo"`
            FCnt  uint32 `json:"fCnt"`
            FPort uint8  `json:"fPort"`
            DeduplicationId string `json:"deduplicationId"`
            Data string `json:"data"`  // base64
            Object map[string]any `json:"object"`  // arbitrary
            RxInfo []struct {
                GatewayId string `json:"gatewayId"`
                Rssi int16 `json:"rssi"`
                Snr float32 `json:"snr"`
                Time string `json:"time"`
            } `json:"rxInfo"`
            Time string `json:"time"`
        }
        if err := json.Unmarshal(payload, &raw); err != nil {
            return Event{}, fmt.Errorf("ingest decode: json: %w", err)
        }
        bytes, _ := base64.StdEncoding.DecodeString(raw.Data)  // empty on error → handled later

        ev := Event{
            DevEUI: strings.ToLower(raw.DeviceInfo.DevEui),
            ApplicationID: raw.DeviceInfo.ApplicationId,
            FCnt: raw.FCnt, FPort: raw.FPort,
            DeduplicationID: raw.DeduplicationId,
            Data: bytes,
            DecodedObject: raw.Object,  // may be nil if codec missing/failed
        }
        for _, r := range raw.RxInfo {
            t := parseRFC3339Maybe(r.Time)
            ev.RxInfo = append(ev.RxInfo, RxInfo{ GatewayID: r.GatewayId, RSSI: r.Rssi, SNR: r.Snr, Time: t })
            if t != nil && (ev.GatewayRxTime == nil || t.Before(*ev.GatewayRxTime)) { ev.GatewayRxTime = t }
        }
        if t := parseRFC3339Maybe(raw.Time); t != nil { ev.DeviceTime = t }
        return ev, nil
    }
    ```

    ```go
        Pool     *pgxpool.Pool
        Resolver *resolver.Resolver  // Plan 02-07
        Mappings MappingStore        // returns mapping rows for a profile
        Profiles ProfileStore        // for counter_modulus lookup
        Log      *slog.Logger

    // UplinkHandler is bound to MQTTSubscriber.SetUplinkHandler at boot
    // (Plan 02-15 cmd/shifter serve wiring). It MUST never panic and SHOULD
    // never silently swallow uplinks — D-26 quality flag persists all rows.
    func UplinkHandler(deps Deps) chirpstack.UplinkHandler {
        return func(topic string, payload []byte) {
            ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
            defer cancel()

            ingestTime := time.Now().UTC()  // DATA-03 — authoritative time

            ev, err := DecodeChirpStackEvent(payload)
            if err != nil {
                // We can't even parse the event — write a row with quality=decode_fail
                // and the raw payload, so operators see the issue. dev_eui MAY be
                // recoverable from the topic (`application/+/device/<eui>/event/up`).
                deps.Log.Warn("ingest: decode failed", "err", err, "topic", topic)
                if devEUI := devEUIFromTopic(topic); devEUI != "" {
                    persistDecodeFail(ctx, deps, devEUI, ingestTime, payload)
                }
                return
            }

            binding, err := deps.Resolver.Lookup(ctx, ev.DevEUI, ingestTime)
            if err != nil {
                if errors.Is(err, resolver.ErrNoActiveBinding) {
                    // Race / unknown device. Persist as quality=missing_canonical so it shows up.
                    deps.Log.Warn("ingest: no active binding", "dev_eui", ev.DevEUI)
                    persistOrphan(ctx, deps, ev, ingestTime, payload, QualityMissingCanonical)
                    return
                }
                deps.Log.Error("ingest: resolver error", "err", err, "dev_eui", ev.DevEUI)
                return
            }

            mappings, err := deps.Mappings.GetMappingsByProfile(ctx, binding.DeviceProfileID)
            if err != nil {
                deps.Log.Error("ingest: load mappings failed", "err", err)
                return
            }

            normalized, normErr := NormalizeMeasurement(ev.DecodedObject, mappings)
            quality := QualityOK
            if normErr != nil {
                quality = QualityMissingCanonical
                deps.Log.Warn("ingest: normalize warning", "dev_eui", ev.DevEUI, "err", normErr)
            }

            // Persist atomically: measurement + last_seen + (rollover) offset advance + audit
            if err := PersistAtomically(ctx, deps, PersistInput{
                IngestTime: ingestTime, Event: ev, Binding: binding, Mappings: mappings,
                Normalized: normalized, RawPayload: payload, Quality: quality,
            }); err != nil {
                deps.Log.Error("ingest: persist failed", "err", err, "dev_eui", ev.DevEUI)
            }
        }
    }
    ```

    ```go
    // NormalizeMeasurement applies a profile's mapping rows against a decoded
    // object, producing the Layer-1 canonical measurement column values + an
    // extra map for vendor-specific fields the mapping promoted to extra.<key>.
    //
    // Per CONTEXT D-01 + D-08 + DATA-09 + Anti-Pattern "no decoders.ts": this
    // function MUST NOT contain a switch on vendor or device type. The mapping
    // table is the only configuration; adding a vendor = adding a profile + its
    // mappings (no Go code change). If this file ever grows a vendor switch,
    // that's a sign the mapping mechanism is being bypassed (PITFALLS §3).

    type Layer1 struct {
        RawValue        *big.Float    // pointed to so nil = missing
        CumulativeValue *big.Float    // = raw + offset (computed by Persist)
        InstantValue    *big.Float    // flow_rate or instant_power
        BatteryPct      *int16
        RSSI            *int16
        SNR             *float32
        TemperatureC    *float32
        PressureKPa     *float32
        LeakDetected    *bool
        TamperDetected  *bool
        Extra           map[string]any  // unfit-for-canonical fields → extra.<key>
    }

    var ErrNoCanonicalValue = errors.New("normalize: no raw_value or instant_value mapped")

    // canonicalTargets enumerates Layer-1 column names that mappings can promote
    // to. Anything not in this list with target prefix "extra." goes to Extra.
    // Anything else is rejected at Plan 02-08 SaveProfile validation.
    var canonicalTargets = map[string]bool{
        "raw_value": true, "cumulative_value": true, "instant_value": true,
        "battery_pct": true, "rssi": true, "snr": true,
        "temperature_c": true, "pressure_kpa": true,
        "leak_detected": true, "tamper_detected": true,
    }

    // NormalizeMeasurement walks each mapping row IN POSITION ORDER and applies
    // it to the decoded object. Returns the populated Layer1 struct + nil error
    // on success. Returns ErrNoCanonicalValue (caller sets quality='missing_canonical')
    // if neither raw_value nor instant_value got populated (one of the two MUST
    // be present for a useful measurement).
    func NormalizeMeasurement(decoded map[string]any, mappings []profile.Mapping) (Layer1, error) {
        out := Layer1{Extra: map[string]any{}}

        for _, m := range mappings {
            val, ok := profile.Resolve(decoded, m.JSONPointer)
            if !ok {
                continue  // missing field — silently skip; quality is set by caller if it matters
            }

            // Coerce per data_type
            coerced, err := coerce(val, m.DataType)
            if err != nil { continue }

            // Apply scale (numeric / int only)
            scaled := applyScale(coerced, m.Scale, m.DataType)

            // Route to canonical column or extra
            if strings.HasPrefix(m.Target, "extra.") {
                key := strings.TrimPrefix(m.Target, "extra.")
                out.Extra[key] = scaled
                continue
            }
            if !canonicalTargets[m.Target] {
                continue  // unknown target — should never happen post-validation
            }
            assignToLayer1(&out, m.Target, scaled)
        }

        if out.RawValue == nil && out.InstantValue == nil {
            return out, ErrNoCanonicalValue
        }
        return out, nil
    }

    func coerce(val any, dataType string) (any, error) { ... }
    func applyScale(val any, scale *big.Float, dataType string) any { ... }
    func assignToLayer1(out *Layer1, target string, val any) { ... }
    ```

    ```go
    type PersistInput struct {
        IngestTime    time.Time
        Event         Event
        Binding       resolver.Binding
        Mappings      []profile.Mapping
        Normalized    Layer1
        RawPayload    []byte
        Quality       string  // QualityOK | QualityMissingCanonical | ...
    }

    // PersistAtomically writes one measurement row + updates device.last_seen +
    // detects rollover (advancing binding.reading_offset and writing an audit
    // row 'rollover_detected' on detection). All in a single Serializable txn.
    //
    // The cumulative_value column is computed here:
    //   cumulative = raw + binding.reading_offset (after rollover application)
    //
    // Per D-26 + Pitfall 5: NEVER silent-drop. Even if Quality != ok, the row
    // gets inserted (with raw_payload + decoded_object preserved). The dashboard
    // surfaces these via the "X uplinks flagged" badge.
    func PersistAtomically(ctx context.Context, deps Deps, in PersistInput) error {
        tx, err := deps.Pool.BeginTx(ctx, pgx.TxOptions{IsoLevel: pgx.Serializable})
        if err != nil { return fmt.Errorf("begin tx: %w", err) }
        defer func() { _ = tx.Rollback(ctx) }()

        q := sqlc.New(tx)

        // 1. Rollover detection — only for canonical raw_value path AND only when
        //    binding.last_raw_value is not nil (skip on first uplink for binding).
        currentOffset := in.Binding.ReadingOffset
        rolloverDetected := false
        if in.Normalized.RawValue != nil && in.Binding.LastRawValue != nil {
            if swap.DetectRollover(in.Binding.LastRawValue, in.Normalized.RawValue) {
                currentOffset = swap.ApplyRollover(in.Binding.ReadingOffset, in.Binding.CounterModulus)
                rolloverDetected = true
                if err := q.AdvanceReadingOffset(ctx, sqlc.AdvanceReadingOffsetParams{
                    ID: in.Binding.BindingID,
                    ReadingOffset: pgNumericFromBigFloat(big.NewFloat(0).SetInt64(in.Binding.CounterModulus)),
                }); err != nil { return fmt.Errorf("advance offset: %w", err) }
            }
        }

        // 2. Compute cumulative_value
        var cumulative *big.Float
        if in.Normalized.RawValue != nil {
            cumulative = new(big.Float).SetPrec(128).Add(in.Normalized.RawValue, currentOffset)
        }

        // 3. Insert measurement row (D-02 + DATA-07 — all 3 layers persisted)
        decodedJSON, _ := json.Marshal(in.Event.DecodedObject)
        extraJSON, _ := json.Marshal(in.Normalized.Extra)
        if err := q.AppendMeasurement(ctx, sqlc.AppendMeasurementParams{
            Time:             pgtype.Timestamptz{Time: in.IngestTime, Valid: true},  // DATA-03
            MeteringPointID:  pgtype.UUID{Bytes: in.Binding.MeteringPointID, Valid: true},
            RawValue:         pgNumericFromBigFloatNullable(in.Normalized.RawValue),
            CumulativeValue:  pgNumericFromBigFloatNullable(cumulative),
    ```
