    ```sql
    -- Mapping rows from decoded JSON to canonical measurement columns (D-08, DATA-09).
    -- json_pointer: RFC 6901 path string (e.g. "/cumulative_l", "/phase_a/voltage").
    -- target: either a canonical measurement column name (e.g. "cumulative_value",
    --        "instant_value", "battery_pct") OR an "extra.<key>" freeform path that
    --        lands in measurement.extra JSONB at <key>.
    -- scale: multiplied with the decoded numeric value (e.g. 0.001 to convert mV→V).
    -- data_type: numeric|int|bool|text — drives type coercion in normalize.go.
    -- ordering: position drives deterministic mapping pass order in normalize engine.
    CREATE TABLE device_profile_mapping (
        id                 UUID PRIMARY KEY DEFAULT gen_random_uuid(),
        device_profile_id  UUID NOT NULL REFERENCES device_profile(id) ON DELETE CASCADE,
        json_pointer       TEXT NOT NULL,
        target             TEXT NOT NULL,
        scale              NUMERIC NOT NULL DEFAULT 1,
        data_type          TEXT NOT NULL,
        position           INT NOT NULL DEFAULT 0,
        created_at         TIMESTAMPTZ NOT NULL DEFAULT now(),
        updated_at         TIMESTAMPTZ NOT NULL DEFAULT now(),
        CONSTRAINT device_profile_mapping_data_type_valid
            CHECK (data_type IN ('numeric','int','bool','text')),
        CONSTRAINT device_profile_mapping_pointer_starts_with_slash
            CHECK (json_pointer = '' OR json_pointer LIKE '/%'),
        CONSTRAINT device_profile_mapping_target_not_empty
            CHECK (length(target) > 0),
        CONSTRAINT device_profile_mapping_unique_target_per_profile
            UNIQUE (device_profile_id, target)
    );

    CREATE INDEX device_profile_mapping_profile_idx ON device_profile_mapping (device_profile_id, position);

    CREATE TRIGGER device_profile_mapping_touch
        BEFORE UPDATE ON device_profile_mapping
        FOR EACH ROW
        EXECUTE FUNCTION touch_updated_at();
    ```

    ```sql
    -- DATA-02: each device-to-metering-point binding has [valid_from, valid_to)
    -- window + reading_offset. D-14 sets valid_to = swap.confirm_time on swap.
    -- D-25 invalidates resolver via NOTIFY binding_changed (trigger added in Plan 07).
    -- Open Q #1 resolution: half-open interval [valid_from, valid_to) — gateway_rx_time
    -- exactly == valid_to is attributed to the NEW binding.

    CREATE EXTENSION IF NOT EXISTS btree_gist;

    CREATE TABLE binding (
        id                 UUID PRIMARY KEY DEFAULT gen_random_uuid(),
        metering_point_id  UUID NOT NULL REFERENCES metering_point(id) ON DELETE RESTRICT,
        device_id          UUID NOT NULL REFERENCES device(id) ON DELETE RESTRICT,
        valid_from         TIMESTAMPTZ NOT NULL,
        valid_to           TIMESTAMPTZ NULL,
        reading_offset     NUMERIC NOT NULL DEFAULT 0,
        last_raw_value     NUMERIC NULL,
        created_at         TIMESTAMPTZ NOT NULL DEFAULT now(),
        CONSTRAINT binding_window_valid CHECK (valid_to IS NULL OR valid_to > valid_from)
    );

    -- Resolver hot-path: which MP is dev_eui currently bound to?
    -- (Phase 4 SSE will reuse the same channel for live KPI updates.)
    CREATE INDEX binding_device_active_idx ON binding (device_id, valid_from DESC);
    CREATE INDEX binding_mp_active_idx     ON binding (metering_point_id, valid_from DESC);

    -- Pitfall 10 mitigation + RESEARCH Pattern 3:
    -- At most ONE active binding per metering_point (no half-open overlap).
    ALTER TABLE binding ADD CONSTRAINT binding_no_overlap_per_mp
        EXCLUDE USING gist (
            metering_point_id WITH =,
            tstzrange(valid_from, COALESCE(valid_to, 'infinity'::timestamptz), '[)') WITH &&
        );
    ```

    NOTE on `last_raw_value`: stored on binding (not device or measurement) so the rollover detector in Plan 02-09 can SELECT one row to fetch the per-binding previous raw counter without a JOIN to a 1B-row hypertable. This column is updated on every successful uplink ingest.

    NOTE: NO trigger emitting NOTIFY binding_changed in this migration. Plan 02-07 (resolver) adds the trigger as part of its package setup so the NOTIFY+LISTEN are wired together. Listing it here would create a NOTIFY with no listener for the days between Plan 03 and Plan 07 (harmless but noisy in logs).
