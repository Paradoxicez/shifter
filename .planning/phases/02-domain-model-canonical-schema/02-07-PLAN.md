        after, err := marshalDiff(e.After)

    // ChangedFields returns the subset of (before -> after) where values differ.
    // EXPLICIT-NULL semantics (Open Q#4): if a field is set in `before` but
    // missing or nil in `after`, the result includes the field with value nil
    // (Go) which marshals to JSON `null`. This distinguishes "field changed
    // FROM something TO null" from "field wasn't part of the diff."
    //
    // Used by handlers to build Entry.Before/After. CREATE → caller passes
    // nil for before and the full insert map for after. UPDATE → caller
    // passes ChangedFields(prev, new).
    func ChangedFields(before, after map[string]any) (beforeDiff, afterDiff map[string]any) {
        beforeDiff = map[string]any{}
        afterDiff = map[string]any{}
        for k, av := range after {
            bv, present := before[k]
            if !present || !reflect.DeepEqual(bv, av) {
                if present { beforeDiff[k] = bv } else { beforeDiff[k] = nil }
                afterDiff[k] = av
            }
        }
        // Catch fields removed in after (cleared)
        for k, bv := range before {
            if _, present := after[k]; !present {
                beforeDiff[k] = bv
                afterDiff[k] = nil  // explicit null per Open Q#4
            }
        }
        return
    }

    -- D-25: resolver listens on binding_changed channel; payload = dev_eui (lowercase).
    -- Trigger fires on (a) INSERT (new binding opened) and (b) UPDATE of valid_to
    -- (binding closed at swap). Lookups for the OLD device_id and NEW device_id
    -- both need invalidation; we emit one NOTIFY per affected dev_eui.

    CREATE OR REPLACE FUNCTION binding_notify_changed() RETURNS trigger AS $$
    DECLARE
        deveui_text TEXT;
    BEGIN
        IF TG_OP = 'INSERT' THEN
            SELECT dev_eui INTO deveui_text FROM device WHERE id = NEW.device_id;
            PERFORM pg_notify('binding_changed', deveui_text);
        ELSIF TG_OP = 'UPDATE' THEN
            -- Only fire when valid_to flips from NULL → ts (binding closed).
            IF OLD.valid_to IS NULL AND NEW.valid_to IS NOT NULL THEN
                SELECT dev_eui INTO deveui_text FROM device WHERE id = NEW.device_id;
                PERFORM pg_notify('binding_changed', deveui_text);
            END IF;
        END IF;
        RETURN NEW;
    END;
    $$ LANGUAGE plpgsql;

    CREATE TRIGGER binding_notify_after_insert
        AFTER INSERT ON binding
        FOR EACH ROW EXECUTE FUNCTION binding_notify_changed();

    CREATE TRIGGER binding_notify_after_update
        AFTER UPDATE OF valid_to ON binding
        FOR EACH ROW EXECUTE FUNCTION binding_notify_changed();
    ```

    // Package swap implements the offset/rollover math required by DATA-02 +
    // DATA-04 + DATA-05. The math is pure: display = raw + offset; on swap,
    // offset is chosen so that the new meter's first cumulative_value equals
    // the outgoing meter's last cumulative_value.
    package swap

    import "math/big"

    // ProposeOffset computes the reading_offset for a new binding such that
    // cumulative_value remains continuous across the swap.
    //
    //   display_old_at_swap = raw_old + offset_old        (the visible reading at swap time, R)
    //   display_new_at_first_uplink = raw_new + offset_new   (must equal R for continuity)
    //   raw_new = N (the new meter's initial counter value)
    //   ⇒ offset_new = R - N
    //
    // Operands are NUMERIC (arbitrary precision) — use math/big.Float (or
    // string-encoded big.Rat if precision really matters — for v1, big.Float
    // with 128-bit precision is more than enough for any plausible meter).
    func ProposeOffset(R, N *big.Float) *big.Float {
        out := new(big.Float).SetPrec(128)
        return out.Sub(R, N)
    }

    // DetectRollover returns true when the device counter wrapped between
    // consecutive uplinks: curr < prev (PITFALLS §2). The caller must NOT call
    // DetectRollover at a binding boundary (the new device's first uplink will
    // generally have curr < prev because raw counters reset). The ingest
    // pipeline (Plan 02-09) handles the binding-boundary case by skipping
    // rollover detection on the first uplink for a freshly opened binding
    // (binding.last_raw_value IS NULL).
    func DetectRollover(prev, curr *big.Float) bool {
        return curr.Cmp(prev) < 0
    }

    // ApplyRollover bumps the binding's reading_offset by counter_modulus
    // (the maximum value the raw counter can hold + 1, e.g. 2^32 for 32-bit
    // counters per D-05).
    //   new_offset = old_offset + counter_modulus
    // Caller (ingest pipeline) writes the new offset to the binding row via
    // sqlc AdvanceReadingOffset and emits an audit_log entry with
    // action='rollover_detected'.
    func ApplyRollover(currentOffset *big.Float, counterModulus int64) *big.Float {
        m := new(big.Float).SetInt64(counterModulus)
        return new(big.Float).SetPrec(128).Add(currentOffset, m)
    }
    ```

    type Deps struct {
        Pool     *pgxpool.Pool
        Resolver Invalidator
        Log      *slog.Logger
    }

    type Invalidator interface {
        Invalidate(devEUI string)
    }

    type SwapInput struct {
        UserID            uuid.UUID
        RequestID         string
        MeteringPointID   uuid.UUID
        OutgoingBindingID uuid.UUID  // current active; will be closed
        OutgoingDevEUI    string     // for resolver invalidation
        IncomingDeviceID  uuid.UUID  // newly bound
        IncomingDevEUI    string
        ConfirmTime       time.Time  // operator click — D-14
        OutgoingReadingR  *big.Float // captured outgoing reading (D-12)
        IncomingInitialN  *big.Float // new meter starting value (often 0)
        OperatorOverride  *big.Float // optional: operator can override proposed offset (D-13)
        OperatorNotes     string     // free text, lands in audit_log.notes
    }

    func CommitSwap(ctx context.Context, deps Deps, in SwapInput) (newBindingID uuid.UUID, err error) {
        // Compute proposed offset (or use override if operator set one).
        proposed := ProposeOffset(in.OutgoingReadingR, in.IncomingInitialN)
        appliedOffset := proposed
        if in.OperatorOverride != nil {
            appliedOffset = in.OperatorOverride
        }

        tx, err := deps.Pool.BeginTx(ctx, pgx.TxOptions{IsoLevel: pgx.Serializable})
        if err != nil { return uuid.Nil, fmt.Errorf("begin tx: %w", err) }
        defer func() { _ = tx.Rollback(ctx) }()

        q := sqlc.New(tx)

        // 1. Close outgoing binding at swap.confirm_time (D-14).
        oldBinding, err := q.CloseBinding(ctx, sqlc.CloseBindingParams{
            ID: in.OutgoingBindingID, ValidTo: pgtype.Timestamptz{Time: in.ConfirmTime, Valid: true},
        })
        if err != nil { return uuid.Nil, fmt.Errorf("close outgoing binding: %w", err) }

        // 2. Open incoming binding (EXCLUDE constraint enforces no overlap).
        offsetNumeric := pgNumericFromBigFloat(appliedOffset)
        newBinding, err := q.OpenBinding(ctx, sqlc.OpenBindingParams{
            MeteringPointID: pgtype.UUID{Bytes: in.MeteringPointID, Valid: true},
            DeviceID:        pgtype.UUID{Bytes: in.IncomingDeviceID, Valid: true},
            ValidFrom:       pgtype.Timestamptz{Time: in.ConfirmTime, Valid: true},
            ReadingOffset:   offsetNumeric,
        })
        if err != nil { return uuid.Nil, fmt.Errorf("open incoming binding: %w", err) }

        // 3. Audit row INSIDE the same txn (D-23).
        before := map[string]any{
            "binding_id":     oldBinding.ID,
            "device_id":      oldBinding.DeviceID,
            "valid_to":       nil,  // before close
            "reading_offset": oldBinding.ReadingOffset,
        }
        after := map[string]any{
            "binding_id":     newBinding.ID,
            "device_id":      in.IncomingDeviceID,
            "valid_from":     in.ConfirmTime,
            "reading_offset": appliedOffset.Text('g', 32),
            "outgoing_R":     in.OutgoingReadingR.Text('g', 32),
            "incoming_N":     in.IncomingInitialN.Text('g', 32),
            "proposed":       proposed.Text('g', 32),
            "override_used":  in.OperatorOverride != nil,
        }
        if err := audit.WriteEntry(ctx, tx, audit.Entry{
            UserID: in.UserID, RequestID: in.RequestID,
            Action: audit.ActionSwap, EntityType: audit.EntityTypeBinding,
            EntityID: newBinding.ID, Before: before, After: after, Notes: in.OperatorNotes,
        }); err != nil {
            return uuid.Nil, fmt.Errorf("write audit: %w", err)
        }

        if err := tx.Commit(ctx); err != nil {
            return uuid.Nil, fmt.Errorf("commit swap: %w", err)
        }

        // 4. Invalidate resolver — defense in depth on top of the LISTEN/NOTIFY
        //    trigger. Direct call ensures the cache is clean even if the
        //    listener is reconnecting after a network blip.
        if deps.Resolver != nil {
            deps.Resolver.Invalidate(strings.ToLower(in.OutgoingDevEUI))
            deps.Resolver.Invalidate(strings.ToLower(in.IncomingDevEUI))
        }
        deps.Log.Info("swap committed",
            "metering_point_id", in.MeteringPointID,
            "old_dev_eui", in.OutgoingDevEUI, "new_dev_eui", in.IncomingDevEUI,
            "applied_offset", appliedOffset.Text('g', 32))
        return newBinding.ID, nil
    }

    // its reading offset, its profile's counter_modulus, etc. Returned by Lookup.
    type Binding struct {
        BindingID         uuid.UUID
        MeteringPointID   uuid.UUID
        DeviceID          uuid.UUID
        ReadingOffset     *big.Float
        LastRawValue      *big.Float  // nil = never received; first uplink for this binding
        CounterModulus    int64
        DeviceProfileID   uuid.UUID
        ValidFrom         time.Time

    // with NO TTL — invalidation comes via NOTIFY binding_changed (D-25 +
    // PITFALLS §2 mitigation).
    type Resolver struct {
        loader     Loader  // injected: usually q.GetActiveBindingByDevEUI wrapper
        mu         sync.RWMutex
        cache      map[string]Binding
        hits       atomic.Int64
        misses     atomic.Int64
        invalids   atomic.Int64
    }

    type Loader interface {
        LoadActive(ctx context.Context, devEUI string, at time.Time) (Binding, error)
    }

    var ErrNoActiveBinding = errors.New("resolver: no active binding for dev_eui at requested time")

    func New(loader Loader) *Resolver { return &Resolver{loader: loader, cache: map[string]Binding{}} }

    // Lookup returns the active binding for devEUI at time t. Cache hit → O(1).
    // Cache miss → DB load via loader; cached on success.
    func (r *Resolver) Lookup(ctx context.Context, devEUI string, at time.Time) (Binding, error) {
        devEUI = strings.ToLower(devEUI)
        r.mu.RLock()
        if b, ok := r.cache[devEUI]; ok {
            // Validate cached entry's window still covers `at` (defensive — handles
            // race where we load just before binding_changed arrives).
            if !at.Before(b.ValidFrom) {
                r.mu.RUnlock()
                r.hits.Add(1)
                return b, nil
            }
        }
        r.mu.RUnlock()

        b, err := r.loader.LoadActive(ctx, devEUI, at)
        if err != nil {
            r.misses.Add(1)
            return Binding{}, err
        }
        r.mu.Lock()
        r.cache[devEUI] = b
        r.mu.Unlock()
        r.misses.Add(1)
        return b, nil
    }

    // and by the ingest pipeline after successful UpdateBindingLastRaw to avoid
    // a re-load on the next uplink for the same dev_eui.
    func (r *Resolver) Set(devEUI string, b Binding) { ... }

    // Invalidate removes a dev_eui from the cache. Implements the Invalidator
    // interface used by swap.CommitSwap.
    func (r *Resolver) Invalidate(devEUI string) {
        r.mu.Lock(); delete(r.cache, strings.ToLower(devEUI)); r.mu.Unlock()
        r.invalids.Add(1)
    }

    func (r *Resolver) Run(ctx context.Context, pool *pgxpool.Pool, log *slog.Logger) {
        for {
            if ctx.Err() != nil { return }
            if err := r.runOnce(ctx, pool, log); err != nil {
                if ctx.Err() != nil { return }
                log.Warn("resolver listener disconnected, reconnecting", "err", err)
                time.Sleep(2 * time.Second)
            }
        }
    }

    func (r *Resolver) runOnce(ctx context.Context, pool *pgxpool.Pool, log *slog.Logger) error {
        conn, err := pool.Acquire(ctx)
        if err != nil { return fmt.Errorf("acquire: %w", err) }
        defer conn.Release()

        if _, err := conn.Exec(ctx, "LISTEN binding_changed"); err != nil {
            return fmt.Errorf("LISTEN: %w", err)
        }
        log.Info("resolver listening on binding_changed")

        for {
            notif, err := conn.Conn().WaitForNotification(ctx)
            if err != nil { return fmt.Errorf("WaitForNotification: %w", err) }
            payload := strings.ToLower(notif.Payload)
            r.Invalidate(payload)
            log.Debug("resolver invalidated", "dev_eui", payload)
        }
    }
