// internal/chirpstack/device.go (skeleton — Pattern 4)
import "github.com/chirpstack/chirpstack/api/go/v4/api"

type CreateDeviceInput struct {
    DevEUI          string  // lowercase 16-char hex
    ApplicationID   string  // CS application UUID
    DeviceProfileID string  // CS device-profile UUID
    Name            string
    Description     string
    AppKey          string  // hex 32 chars
    JoinEUI         string  // hex 16 chars; default "0000000000000000" for 1.0.x
}

func (c *Client) CreateDevice(ctx context.Context, in CreateDeviceInput) error {
    devSvc := api.NewDeviceServiceClient(c.conn)
    _, err := devSvc.Create(ctx, &api.CreateDeviceRequest{
        Device: &api.Device{
            DevEui:          in.DevEUI,
            ApplicationId:   in.ApplicationID,
            DeviceProfileId: in.DeviceProfileID,
            Name:            in.Name,
            Description:     in.Description,
        },
    })
    if err != nil { return fmt.Errorf("DeviceService.Create: %w", err) }

    _, err = devSvc.CreateKeys(ctx, &api.CreateDeviceKeysRequest{
        DeviceKeys: &api.DeviceKeys{
            DevEui: in.DevEUI,
            AppKey: in.AppKey,
            NwkKey: in.AppKey, // 1.0.x compat (CS server-side handles split for 1.1.x)
        },
    })
    if err != nil { return fmt.Errorf("DeviceService.CreateKeys: %w", err) }
    return nil
}
```

func (c *Client) EnsureTenant(ctx context.Context, name string) (string, error) {
    svc := api.NewTenantServiceClient(c.conn)
    list, err := svc.List(ctx, &api.ListTenantsRequest{Limit: 100, Search: name})
    if err == nil {
        for _, t := range list.Result { if t.Name == name { return t.Id, nil } }
    }
    resp, err := svc.Create(ctx, &api.CreateTenantRequest{
        Tenant: &api.Tenant{
            Name: name, Description: "Shifter-managed tenant — do not edit",
            CanHaveGateways: true, PrivateGatewaysUp: true, PrivateGatewaysDown: true,
        },
    })
    if err != nil { return "", fmt.Errorf("TenantService.Create: %w", err) }
    return resp.Id, nil
}

    package chirpstack

    // EnsureTenantAndApplication is the D-28 first-boot routine. Reads
    // chirpstack_connection.cs_tenant_id + cs_application_id; if either is NULL,
    // creates / reuses the corresponding ChirpStack entity and writes the
    // returned UUID back. Subsequent boots short-circuit (both UUIDs already
    // populated → return without calling CS).
    //
    // Failure to reach ChirpStack is logged as a WARNING (slog) and returned as
    // an error — caller (cmd/shifter serve) decides whether to refuse boot or
    // continue with degraded functionality. Per Pitfall 9, we recommend NOT
    // blocking serve startup on this — if CS is temporarily unreachable, the
    // resolver + ingest code will still work for already-bound devices once CS
    // returns; first add-device after recovery will trigger another bootstrap
    // attempt.
    type ConnectionStore interface {
        GetCSConnection(ctx context.Context) (cs_tenant_id, cs_application_id string, err error)
        SetCSTenantApp(ctx context.Context, tenantID, applicationID string) error
    }

    func EnsureTenantAndApplication(ctx context.Context, c *Client, store ConnectionStore, log *slog.Logger) (tenantID, appID string, err error) {
        cur_t, cur_a, err := store.GetCSConnection(ctx)
        if err != nil { return "", "", fmt.Errorf("read chirpstack_connection: %w", err) }

        if cur_t != "" && cur_a != "" {
            log.Debug("chirpstack bootstrap skipped — already bootstrapped",
                "cs_tenant_id", cur_t, "cs_application_id", cur_a)
            return cur_t, cur_a, nil
        }

        // Tenant: ensure or reuse "shifter-default" (D-28).
        tenantName := "shifter-default"
        if cur_t == "" {
            cur_t, err = c.EnsureTenant(ctx, tenantName)
            if err != nil { return "", "", fmt.Errorf("ensure tenant %q: %w", tenantName, err) }
        }

        // Application: ensure or reuse "shifter" under that tenant.
        appName := "shifter"
        if cur_a == "" {
            cur_a, err = c.EnsureApplication(ctx, cur_t, appName)
            if err != nil { return "", "", fmt.Errorf("ensure application %q: %w", appName, err) }
        }

        if err := store.SetCSTenantApp(ctx, cur_t, cur_a); err != nil {
            return "", "", fmt.Errorf("persist cs ids: %w", err)
        }
        log.Info("chirpstack bootstrap complete",
            "cs_tenant_id", cur_t, "cs_application_id", cur_a)
        return cur_t, cur_a, nil
    }
    ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Shifter ↔ ChirpStack gRPC | API token stored by ref (Phase 1 D-22); calls flow over TLS by default; bufconn mock used in tests |
| Codec_js ↔ ChirpStack QuickJS | Operator-pasted codec runs in CS sandbox (D-09 + RESEARCH §Security "Codec_js injection"); never executes in Shifter process |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-05-01 | I (Information Disclosure) | API token in logs | mitigate | Phase 1 D-22 already locked: token never logged; wrappers don't pass token in error wrapping (only the error from gRPC layer which doesn't include auth header) |
| T-02-05-02 | T (Tampering) | codec_js injection | mitigate | Codec runs in ChirpStack QuickJS sandbox — no access to Shifter process; admin-only RBAC (Phase 1 + Plan 02-10 auth.Can extension); length cap enforced in Plan 02-08 profile editor |
