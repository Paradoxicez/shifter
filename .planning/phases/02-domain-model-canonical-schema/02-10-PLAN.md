    ```go
    // ParseDevEUI implements D-11: parse a hex string from a vendor sticker
    // that may use either MSB-first or LSB-first byte order. Returns BOTH
    // interpretations + vendor OUI hint (first 3 bytes ↔ IEEE OUI assignment).
    // The UI presents both side-by-side and the operator clicks the right one.
    type DevEUIPreview struct {
        MSB        string  // hex16 lowercase as pasted (after whitespace strip)
        LSB        string  // hex16 lowercase, byte-reversed
        MSBVendor  string  // OUI lookup hint or "unknown"
        LSBVendor  string
    }

    var ErrBadDevEUI = errors.New("DevEUI must be 16 hex characters after whitespace strip")

    // Whitespace, hyphens, colons all stripped. Output is lowercase.
    func ParseDevEUI(raw string) (DevEUIPreview, error) {
        cleaned := strings.Map(func(r rune) rune {
            switch r { case ' ', '-', ':', '\t', '\n', '\r': return -1 }
            return unicode.ToLower(r)
        }, raw)
        if len(cleaned) != 16 { return DevEUIPreview{}, ErrBadDevEUI }
        if _, err := hex.DecodeString(cleaned); err != nil { return DevEUIPreview{}, ErrBadDevEUI }
        msb := cleaned
        // Reverse bytes (each pair swapped):
        b, _ := hex.DecodeString(msb)
        for i, j := 0, len(b)-1; i < j; i, j = i+1, j-1 { b[i], b[j] = b[j], b[i] }
        lsb := hex.EncodeToString(b)
        return DevEUIPreview{
            MSB: msb, LSB: lsb,
            MSBVendor: ouiVendor(msb[:6]),
            LSBVendor: ouiVendor(lsb[:6]),
        }, nil
    }

    // ouiVendor — minimal IEEE OUI lookup for Phase 2's 3 vendor families.
    // Phase 7 catalog will expand this. Unknown OUI → "unknown".
    func ouiVendor(oui6hex string) string {
        switch strings.ToLower(oui6hex) {
        case "70b3d5":  // IEEE block including Axioma allocations
            return "Axioma (or other 70:B3:D5 lessee)"
        case "a84041":  // Acrel-allocated block (verify at execution time against IEEE OUI registry)
            return "Acrel"
        }
        return "unknown"
    }
    ```

    ```go
    type AddDeviceRequest struct {
        DevEUI          string  // lowercase 16 hex (UI sends after parser)
        Name            string
        DeviceProfileID uuid.UUID
        AppKey          string  // 32 hex chars — sent to CS, never stored locally
        JoinEUI         string  // optional, default "0000000000000000"
        MeteringPointID uuid.UUID  // if set, opens initial binding in same tx (D-10 step 3)
        InitialReading  *string  // optional starting reading for the binding (becomes reading_offset)
    }

    func addDevice(deps Deps) http.HandlerFunc {
        return func(w http.ResponseWriter, r *http.Request) {
            user, _ := auth.UserFromContext(r.Context())
            if !auth.Can(user, "device.add", nil) {
                http.Error(w, "forbidden", http.StatusForbidden); return
            }

            var in AddDeviceRequest
            if err := json.NewDecoder(r.Body).Decode(&in); err != nil { http.Error(w, "bad request", 400); return }

            // 1. Server-side validate DevEUI format + AppKey format
            if !isHex(in.DevEUI, 16) { http.Error(w, "invalid dev_eui", 400); return }
            if !isHex(in.AppKey, 32) { http.Error(w, "invalid AppKey", 400); return }
            if in.JoinEUI == "" { in.JoinEUI = "0000000000000000" }
            if !isHex(in.JoinEUI, 16) { http.Error(w, "invalid JoinEUI", 400); return }

            // 2. Bootstrap CS tenant + application if not yet (D-28 idempotent).
            tenantID, appID, err := deps.Bootstrap(r.Context())  // wraps chirpstack.EnsureTenantAndApplication
            if err != nil {
                http.Error(w, "ChirpStack unreachable: "+err.Error(), http.StatusBadGateway); return
            }

            // 3. Get profile for cs_profile_id (must be already synced — Plan 02-08 seed sync)
            prof, err := deps.Q.GetDeviceProfile(r.Context(), in.DeviceProfileID)
            if err != nil { http.Error(w, "profile not found", 404); return }
            if !prof.CsProfileID.Valid {
                http.Error(w, "profile not synced to ChirpStack — open Settings → Sync profiles", http.StatusBadGateway); return
            }

            // 4. Atomic transaction — if any step fails, rollback Postgres AND attempt to delete the CS device (best-effort cleanup).
            tx, err := deps.Pool.BeginTx(r.Context(), pgx.TxOptions{IsoLevel: pgx.Serializable})
            if err != nil { http.Error(w, "tx begin failed", 500); return }
            defer func() { _ = tx.Rollback(r.Context()) }()

            // 4a. CS Create + CreateKeys (D-10 step 4)
            csInput := chirpstack.CreateDeviceInput{
                DevEUI: in.DevEUI, ApplicationID: appID, DeviceProfileID: prof.CsProfileID.String,
                Name: in.Name, AppKey: in.AppKey, JoinEUI: in.JoinEUI,
            }
            if err := deps.CS.CreateDevice(r.Context(), csInput); err != nil {
                http.Error(w, "ChirpStack rejected the device: "+err.Error(), http.StatusBadGateway); return
            }
            // Per D-16: on Postgres failure below, delete the CS device for best-effort consistency.
            csCreated := true

            q := sqlc.New(tx)

            // 4b. Insert Shifter device row.
            dev, err := q.CreateDevice(r.Context(), sqlc.CreateDeviceParams{
                DevEui: in.DevEUI, Name: in.Name,
                DeviceProfileID: pgtype.UUID{Bytes: in.DeviceProfileID, Valid: true},
                JoinEui: pgtype.Text{String: in.JoinEUI, Valid: true},
            })
            if err != nil {
                if csCreated { _ = deps.CS.DeleteDevice(r.Context(), in.DevEUI) }
                http.Error(w, "device row insert failed: "+err.Error(), 500); return
            }

            // 4c. Open initial binding if MP picked.
            var bindingID uuid.UUID
            if in.MeteringPointID != uuid.Nil {
                offsetBig := big.NewFloat(0)
                if in.InitialReading != nil {
                    if f, _, err := big.ParseFloat(*in.InitialReading, 10, 128, big.ToNearestEven); err == nil {
                        offsetBig = f
                    }
                }
                b, err := q.OpenBinding(r.Context(), sqlc.OpenBindingParams{
                    MeteringPointID: pgtype.UUID{Bytes: in.MeteringPointID, Valid: true},
                    DeviceID: dev.ID,
                    ValidFrom: pgtype.Timestamptz{Time: time.Now().UTC(), Valid: true},
                    ReadingOffset: pgNumericFromBigFloat(offsetBig),
                })
                if err != nil {
                    if csCreated { _ = deps.CS.DeleteDevice(r.Context(), in.DevEUI) }
                    http.Error(w, "open initial binding failed: "+err.Error(), 500); return
                }
                bindingID = b.ID.Bytes
            }

            // 4d. Audit row inside same tx (D-23).
            after := map[string]any{ "dev_eui": in.DevEUI, "name": in.Name, "device_profile_id": in.DeviceProfileID }
            if in.MeteringPointID != uuid.Nil {
                after["initial_binding"] = map[string]any{ "metering_point_id": in.MeteringPointID, "binding_id": bindingID, "reading_offset": in.InitialReading }
            }
            if err := audit.WriteEntry(r.Context(), tx, audit.Entry{
                UserID: user.ID, RequestID: middleware.GetReqID(r.Context()),
                Action: audit.ActionCreate, EntityType: audit.EntityTypeDevice,
                EntityID: dev.ID.Bytes, After: after,
            }); err != nil {
                if csCreated { _ = deps.CS.DeleteDevice(r.Context(), in.DevEUI) }
                http.Error(w, "audit write failed: "+err.Error(), 500); return
            }

            if err := tx.Commit(r.Context()); err != nil {
                if csCreated { _ = deps.CS.DeleteDevice(r.Context(), in.DevEUI) }
                http.Error(w, "commit failed: "+err.Error(), 500); return
            }

            json.NewEncoder(w).Encode(dev)
        }
    }
    ```
