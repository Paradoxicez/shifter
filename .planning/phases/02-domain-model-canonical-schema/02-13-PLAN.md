ChirpStack v4 uplink event JSON shape (Pitfall 11 — object is map, not string):
```json
{
  "deviceInfo": { "devEui": "0102030405060708", "applicationId": "<uuid>" },
  "fCnt": 42,
  "fPort": 1,
  "data": "<base64 raw bytes>",
  "object": { "cumulative_l": 12345, "battery_pct": 87, "temperature_c": 23.5, "leak": false, "tamper": false },
  "rxInfo": [{ "time": "2026-05-04T12:00:00Z", ... }],
  "time": "2026-05-04T11:59:58Z"
}
```
The `object` field is what the codec produced; `data` is the original raw bytes. Both are required by ingest's decode.go. The synthesizer can fabricate plausible raw bytes (zero-filled or codec-input-shaped) — what matters for the integration tests is that `object` matches the seeded mapping rows so normalize.go produces canonical values.

       // Package testharness — synthetic uplink publisher.
       package testharness

       import (
           "context"
           "encoding/json"
           "fmt"
           "time"

           paho "github.com/eclipse/paho.mqtt.golang"
       )

       type Publisher struct {
           c paho.Client
       }

       // NewPublisher connects to brokerURL (e.g. "tcp://localhost:1883") with
       // optional username/password. Returns once CONNACK is received or fails
       // after 5s.
       func NewPublisher(brokerURL, user, pass, clientID string) (*Publisher, error) {
           opts := paho.NewClientOptions().AddBroker(brokerURL).SetClientID(clientID)
           if user != "" { opts.SetUsername(user) }
           if pass != "" { opts.SetPassword(pass) }
           opts.SetConnectTimeout(5 * time.Second)
           c := paho.NewClient(opts)
           tk := c.Connect()
           if !tk.WaitTimeout(5 * time.Second) {
               return nil, fmt.Errorf("connect timeout")
           }
           if err := tk.Error(); err != nil {
               return nil, err
           }
           return &Publisher{c: c}, nil
       }

       // PublishUplink publishes the given event JSON on the canonical
       // application/<appID>/device/<devEUI>/event/up topic at QoS=1.
       func (p *Publisher) PublishUplink(ctx context.Context, applicationID, devEUI string, event []byte) error {
           topic := fmt.Sprintf("application/%s/device/%s/event/up", applicationID, devEUI)
           tk := p.c.Publish(topic, 1, false, event)
           done := make(chan error, 1)
           go func() {
               tk.Wait()
               done <- tk.Error()
           }()
           select {
           case <-ctx.Done():
               return ctx.Err()
           case err := <-done:
               return err
           }
       }

       func (p *Publisher) Close() {
           p.c.Disconnect(250)
       }
       ```

       package testharness

       import (
           "encoding/base64"
           "encoding/binary"
           "encoding/json"
           "time"
       )

       // BuildAxiomaW1Uplink returns the v4 event JSON for a synthetic Axioma
       // Qalcosonic W1 uplink. The decoded `object` keys MATCH axioma_w1.js
       // result.* exactly (cumulative_l, battery_pct, temperature_c, leak,
       // tamper) per the pinned table in 02-13-PLAN.md.
       //
       // rawCounter is the cumulative water volume counter in liters (per
       // F1 V1.8 codec output spec). Wraps at 2^32-1.
       func BuildAxiomaW1Uplink(rawCounter uint32, batteryPct uint8, tempC int8, leak, tamper bool, fcnt uint32, ingestAt time.Time) ([]byte, error) {
           // Build the raw frame the codec_js would consume — for the
           // INTEGRATION TEST we don't actually run the codec (CS does), so
           // `data` is decorative; what matters is `object`.
           rawBytes := make([]byte, 11)
           // byte 0: status flags
           if tamper { rawBytes[0] |= 0x02 }
           if leak   { rawBytes[0] |= 0x08 }
           // bytes 5-8: cumulative volume LE
           binary.LittleEndian.PutUint32(rawBytes[5:9], rawCounter)
           // byte 9: battery
           rawBytes[9] = batteryPct
           // byte 10: temperature signed int8
           rawBytes[10] = byte(tempC)

           // The decoded `object` keys here MUST match axioma_w1.js result.X.
           // Pinned per `<codec_keys_pinned>` block in 02-13-PLAN.md.
           obj := map[string]any{
               "cumulative_l":   rawCounter,        // result.cumulative_l (axioma_w1.js line 72)
               "battery_pct":    int(batteryPct),   // result.battery_pct  (line 83)
               "temperature_c":  int(tempC),        // result.temperature_c (line 90)
               "leak":           leak,              // result.leak         (line 49)
               "tamper":         tamper,            // result.tamper       (line 47)
           }

           event := map[string]any{
               "deviceInfo": map[string]any{
                   "devEui":        "AXIOMAW1TESTEUI0",  // 16 hex; tests can override
                   "applicationId": "00000000-0000-0000-0000-000000000000",
               },
               "fCnt":   fcnt,
               "fPort":  100,                            // axioma_w1.js requires fPort==100
               "data":   base64.StdEncoding.EncodeToString(rawBytes),
               "object": obj,
               "rxInfo": []map[string]any{
                   {"time": ingestAt.UTC().Format(time.RFC3339Nano)},
               },
               "time": ingestAt.UTC().Format(time.RFC3339Nano),
           }
           return json.Marshal(event)
       }
       ```
