    ```js
    function decodeUplink(input) {
        const bytes = input.bytes;
        const fPort = input.fPort;

        if (fPort !== 100) {
            return { data: {}, errors: ["unexpected fPort " + fPort + ", expected 100"] };
        }

        // Header (4 bytes log time, 1 byte status, 4 bytes log volume LE int32)
        // Following: short packet (1 reading, 11 bytes total) or extended (15+ readings)
        const result = {};
        try {
            // status byte 0
            const status = bytes[0];
            result.battery_low      = (status & 0x04) !== 0;
            result.permanent_error  = (status & 0x10) !== 0;
            result.temporary_error  = (status & 0x20) !== 0;
            result.empty_pipe       = (status & 0x40) !== 0;
            result.reverse_flow     = (status & 0x80) !== 0;
            result.tamper           = (status & 0x02) !== 0;
            result.leak             = (status & 0x08) !== 0;

            // Bytes 1-4: log time (UTC unix-32 LE)
            const logTime = bytes[1] | (bytes[2] << 8) | (bytes[3] << 16) | (bytes[4] << 24);

            // Bytes 5-8: cumulative volume in liters (signed int32 LE)
            const litersRaw = (bytes[5] | (bytes[6] << 8) | (bytes[7] << 16) | (bytes[8] << 24)) >>> 0;
            result.cumulative_l = litersRaw;
            result.cumulative_m3 = litersRaw / 1000.0;

            // Byte 9: battery (% × 2, range 0..200)
            if (bytes.length >= 10) result.battery_pct = Math.min(100, bytes[9]);
            // Byte 10: temperature (signed int8 °C)
            if (bytes.length >= 11) {
                let t = bytes[10]; if (t > 127) t -= 256;
                result.temperature_c = t;
            }

            // Compute flow_rate from delta if extended packet present (15 historical
            // readings spaced 1 hour); for short packet, flow_rate is null and the
            // ingest pipeline computes it from successive uplinks.

            return { data: result, warnings: [], errors: [] };
        } catch (e) {
            return { data: { _raw_hex: bytesToHex(bytes) }, errors: ["decode failed: " + e.message] };
        }
    }

    function bytesToHex(b) {
        let s = ""; for (let i = 0; i < b.length; i++) {
            const h = b[i].toString(16); s += (h.length === 1 ? "0" : "") + h;
        } return s;
    }
    ```

    Skeleton:
    ```js
    // Acrel ADL200 + ADW300 shared LoRaWAN codec.
    //
    // Reference: Acrel ADW300 user manual (linked in 02-RESEARCH.md §"Vendor
    // profile sources"). Payload is a Modbus-derived register-pair stream:
    //   header byte (frame type) + register-count byte + (reg_addr u16 LE +
    //   value u32 LE) × N.
    //
    // Common register addresses (per manual table 4-3):
    //   0x0048: kWh forward (×0.01 kWh)
    //   0x0044: voltage L1-N (×0.1 V)
    //   0x004C: current L1 (×0.001 A)
    //   0x0066: power factor L1 (×0.001)
    //   ... (3-phase devices add L2/L3 versions of all of the above)
    //   0x00B6: battery (% as u8)
    //   0x00C0: temperature (°C signed int8)
    //
    // ADL200 emits a subset (only L1 + battery); ADW300 emits the full set.

    function decodeUplink(input) {
        const bytes = input.bytes;
        const result = {};

        try {
            // Validate frame
            if (bytes.length < 2) return { data: {}, errors: ["payload too short"] };
            const regCount = bytes[1];
            const expectedLen = 2 + regCount * 6; // 2 header + 6 per register pair
            if (bytes.length < expectedLen) {
                return { data: {}, errors: ["truncated frame: expected " + expectedLen + " bytes, got " + bytes.length] };
            }

            for (let i = 0; i < regCount; i++) {
                const off = 2 + i * 6;
                const reg = bytes[off] | (bytes[off + 1] << 8);
                const valU32 = (bytes[off + 2] | (bytes[off + 3] << 8) | (bytes[off + 4] << 16) | (bytes[off + 5] << 24)) >>> 0;
                applyRegister(reg, valU32, result);
            }

            return { data: result, warnings: [], errors: [] };
        } catch (e) {
            return { data: {}, errors: ["decode failed: " + e.message] };
        }
    }

    function applyRegister(reg, val, out) {
        switch (reg) {
            case 0x0048: out.kwh_forward = val * 0.01; break;
            case 0x0044: out.voltage_l1 = val * 0.1; break;
            case 0x004C: out.current_l1 = val * 0.001; break;
            case 0x0066: out.pf_l1 = val * 0.001; break;
            case 0x004A: out.voltage_l2 = val * 0.1; break;
            case 0x004E: out.current_l2 = val * 0.001; break;
            case 0x0068: out.pf_l2 = val * 0.001; break;
            case 0x004B: out.voltage_l3 = val * 0.1; break;
            case 0x004F: out.current_l3 = val * 0.001; break;
            case 0x006A: out.pf_l3 = val * 0.001; break;
            case 0x0040: out.power_total_w = val; break;  // active power in W
            case 0x00B6: out.battery_pct = val; break;
            case 0x00C0: {
                let t = val & 0xff; if (t > 127) t -= 256; out.temperature_c = t; break;
            }
            // Unknown registers go into _unknown so operators can map them via the editor.
            default: {
                if (!out._unknown) out._unknown = {};
                out._unknown["0x" + reg.toString(16).padStart(4, "0")] = val;
            }
        }
    }
    ```

    NOTE: The exact register addresses MAY be wrong in this skeleton. The executor MUST cross-reference the Acrel manual at execution time and adjust. The codec is pushed to ChirpStack on every save, so iteration is cheap.

    ```go
    // Package profile — JSON Pointer (RFC 6901) lookup over a decoded JSON tree.
    // Used by the ingest normalize pass (Plan 02-09) to resolve mapping rows'
    // json_pointer against the decoded_object emitted by the QuickJS codec.

    // Resolve walks the pointer through obj. Returns (value, true) on success.
    // Returns (nil, false) for missing/invalid path. Empty pointer "" returns
    // the root obj.
    //
    // Supports:
    //   ""                    → root
    //   "/foo"                → obj["foo"]
    //   "/foo/bar"            → obj["foo"]["bar"]
    //   "/list/0"             → obj["list"][0]
    //   "/escaped~0tilde"     → obj["escaped~tilde"] (~0 → ~)
    //   "/escaped~1slash"     → obj["escaped/slash"] (~1 → /)
    func Resolve(obj any, ptr string) (any, bool) {
        if ptr == "" { return obj, true }
        if !strings.HasPrefix(ptr, "/") { return nil, false }
        parts := strings.Split(ptr[1:], "/")
        cur := obj
        for _, p := range parts {
            // RFC 6901 escape: ~1 → /, ~0 → ~ (in that order — order matters).
            p = strings.ReplaceAll(p, "~1", "/")
            p = strings.ReplaceAll(p, "~0", "~")

            switch v := cur.(type) {
            case map[string]any:
                next, ok := v[p]
                if !ok { return nil, false }
                cur = next
            case []any:
                idx, err := strconv.Atoi(p)
                if err != nil || idx < 0 || idx >= len(v) { return nil, false }
                cur = v[idx]
            default:
                return nil, false
            }
        }
        return cur, true
    }
    ```

    package profile

    type Mapping struct {
        JSONPointer string  // RFC 6901
        Target      string  // "cumulative_value" or "extra.<key>"
        Scale       *big.Float
        DataType    string  // "numeric" | "int" | "bool" | "text"
        Position    int
    }

    type ProfileSaveInput struct {
        UserID    uuid.UUID
        RequestID string

        // Identity
        ID            uuid.UUID  // zero = create; non-zero = update
        Slug          string     // immutable on update
        Name          string
        Vendor        string
        Family        string
        Capabilities  []string
        CounterModulus int64
        Region         string  // empty = inherit install
        MACVersion     string

        // Codec
        CodecJS        string  // raw QuickJS source

        // Mappings
        Mappings []Mapping  // full set; old mappings deleted + replaced atomically
    }

    type Deps struct {
        Pool      *pgxpool.Pool
        CSClient  CSProfileClient
        ConnStore ConnectionStore  // for tenantID
        Log       *slog.Logger
    }

    type CSProfileClient interface {
        CreateDeviceProfile(ctx context.Context, in chirpstack.CreateProfileInput) (string, error)
        UpdateDeviceProfile(ctx context.Context, in chirpstack.UpdateProfileInput) error
    }

    type ConnectionStore interface {
        GetCSConnection(ctx context.Context) (cs_tenant_id, cs_application_id string, err error)
    }

    const MaxCodecJSBytes = 256 * 1024  // 256 KiB cap (T-08-02 mitigation per Plan 05's threat model)

    func SaveProfile(ctx context.Context, deps Deps, in ProfileSaveInput) (uuid.UUID, error) {
        // 1. Validate codec_js
        if len(in.CodecJS) > MaxCodecJSBytes {
            return uuid.Nil, fmt.Errorf("codec_js exceeds %d bytes", MaxCodecJSBytes)
        }
        if in.CodecJS != "" && !strings.Contains(in.CodecJS, "function decodeUplink") {
            return uuid.Nil, errors.New("codec_js must define a decodeUplink function (ChirpStack QuickJS contract)")
        }
        // Validate capabilities token set (D-04)
        for _, c := range in.Capabilities {
            if !validCapability(c) {
                return uuid.Nil, fmt.Errorf("invalid capability %q (allowed: cumulative,flow_rate,instant_power,battery,temperature,pressure,leak_detection,tamper_detection,multi_phase,power_quality)", c)
            }
        }

        // 2. Get CS tenant id for the gRPC call
        tenantID, _, err := deps.ConnStore.GetCSConnection(ctx)
        if err != nil { return uuid.Nil, fmt.Errorf("read cs connection: %w", err) }
        if tenantID == "" {
            return uuid.Nil, errors.New("ChirpStack tenant not bootstrapped — open Settings → Test connection")
        }

        // 3. Begin txn — Postgres mutations + audit happen here. CS push is OUTSIDE
        //    the txn but BEFORE the commit so failures roll back.
        //    (CS sync failure is handled per Pitfall 9: not a swap-style atomic
        //    requirement; we mark codec_js_synced_at = NULL and the seed routine
        //    retries.)
        tx, err := deps.Pool.BeginTx(ctx, pgx.TxOptions{IsoLevel: pgx.Serializable})
        if err != nil { return uuid.Nil, fmt.Errorf("begin tx: %w", err) }
        defer func() { _ = tx.Rollback(ctx) }()

        q := sqlc.New(tx)

        // 4. Create or update the device_profile row
        var profileID uuid.UUID
        var auditAction string
        if in.ID == uuid.Nil {
            row, err := q.CreateDeviceProfile(ctx, sqlc.CreateDeviceProfileParams{ ... })
            if err != nil { return uuid.Nil, fmt.Errorf("create profile: %w", err) }
            profileID = row.ID
            auditAction = audit.ActionProfileCreate
        } else {
            row, err := q.UpdateDeviceProfile(ctx, sqlc.UpdateDeviceProfileParams{ ID: in.ID, ... })
            if err != nil { return uuid.Nil, fmt.Errorf("update profile: %w", err) }
            profileID = row.ID
            auditAction = audit.ActionProfileUpdate
        }

        // 5. Write codec_js to the row
        if err := q.SetProfileCodecJS(ctx, sqlc.SetProfileCodecJSParams{ ID: profileID, CodecJs: in.CodecJS }); err != nil {
            return uuid.Nil, fmt.Errorf("set codec_js: %w", err)
        }

        // 6. Replace mappings (DELETE all + INSERT new)
        if err := q.DeleteMappingsByProfile(ctx, profileID); err != nil { return uuid.Nil, fmt.Errorf("delete old mappings: %w", err) }
        for _, m := range in.Mappings {
            if err := q.CreateDeviceProfileMapping(ctx, sqlc.CreateDeviceProfileMappingParams{ ... }); err != nil {
                return uuid.Nil, fmt.Errorf("create mapping: %w", err)
            }
        }

        // 7. Audit (D-23 inside txn)
        before := map[string]any{}  // for create, empty
        after := map[string]any{ "name": in.Name, "vendor": in.Vendor, "capabilities": in.Capabilities, "mapping_count": len(in.Mappings) }
        if err := audit.WriteEntry(ctx, tx, audit.Entry{ UserID: in.UserID, RequestID: in.RequestID, Action: auditAction, EntityType: audit.EntityTypeDeviceProfile, EntityID: profileID, Before: before, After: after }); err != nil {
            return uuid.Nil, fmt.Errorf("write audit: %w", err)
        }

        // 8. Push codec_js to ChirpStack (BEFORE commit — failure here rolls back).
        if in.CodecJS != "" {
            row, err := q.GetDeviceProfile(ctx, profileID)
            if err != nil { return uuid.Nil, fmt.Errorf("re-read profile: %w", err) }
            var csID string
            if row.CsProfileID.Valid {
                if err := deps.CSClient.UpdateDeviceProfile(ctx, chirpstack.UpdateProfileInput{ ID: row.CsProfileID.String, Name: in.Name, CodecJS: in.CodecJS }); err != nil {
                    return uuid.Nil, fmt.Errorf("CS UpdateDeviceProfile: %w", err)
                }
                csID = row.CsProfileID.String
            } else {
                csID, err = deps.CSClient.CreateDeviceProfile(ctx, chirpstack.CreateProfileInput{ TenantID: tenantID, Name: in.Name, Region: in.Region, MACVersion: in.MACVersion, CodecJS: in.CodecJS })

    // RunSeedSync iterates ListUnsyncedProfiles (codec_js NULL OR
    // codec_js_synced_at NULL); for each, fills codec_js from //go:embed
    // (CodecBySlug) and pushes to ChirpStack. Idempotent — re-running is a no-op
    // for profiles already synced.
    //
    // Per Pitfall 9, this routine is best-effort: CS reachability failures are
    // logged as warnings and retried on next boot or via Settings → Sync (Phase 6).
    // Boot does NOT block on this — the resolver + ingest still work for
    // already-bound devices once CS returns.
    func RunSeedSync(ctx context.Context, deps Deps) {
        unsynced, err := sqlc.New(deps.Pool).ListUnsyncedProfiles(ctx)
        if err != nil {
            deps.Log.Error("seed sync: list unsynced failed", "err", err)
            return
        }
        if len(unsynced) == 0 {
            deps.Log.Debug("seed sync: nothing to do")
            return
        }

        tenantID, _, err := deps.ConnStore.GetCSConnection(ctx)
        if err != nil || tenantID == "" {
            deps.Log.Warn("seed sync: chirpstack tenant not bootstrapped, retrying on next boot",
                "unsynced_count", len(unsynced), "err", err)
            return
        }

        synced := 0
        for _, p := range unsynced {
            codecBody := codecs.CodecBySlug(p.Slug)
            if codecBody == "" {
                deps.Log.Warn("seed sync: no embedded codec for slug, skipping", "slug", p.Slug)
                continue
            }
            // Build a SaveProfile-equivalent flow per slug — but skip the editor's
            // mapping rewrite (mappings come from a separate migration in Phase 7
            // when full seed mappings ship; for v1 the operator-facing editor
            // populates mappings on first edit).
            csInput := chirpstack.CreateProfileInput{
                TenantID: tenantID, Name: p.Name,
                Region: p.Region.String, MACVersion: p.MacVersion,
                CodecJS: codecBody,
            }
            var csID string
            if p.CsProfileID.Valid {
                csID = p.CsProfileID.String
                if err := deps.CSClient.UpdateDeviceProfile(ctx, chirpstack.UpdateProfileInput{
                    ID: csID, Name: p.Name, CodecJS: codecBody,
                }); err != nil {
                    deps.Log.Warn("seed sync: CS Update failed, retrying next boot",
                        "slug", p.Slug, "err", err)
                    continue
                }
            } else {
                id, err := deps.CSClient.CreateDeviceProfile(ctx, csInput)
                if err != nil {
                    deps.Log.Warn("seed sync: CS Create failed, retrying next boot",
                        "slug", p.Slug, "err", err)
                    continue
                }
                csID = id
            }
            // Persist codec_js + cs_profile_id + sync timestamp in one txn.
            tx, err := deps.Pool.BeginTx(ctx, pgx.TxOptions{IsoLevel: pgx.Serializable})
            if err != nil {
                deps.Log.Error("seed sync: begin tx failed", "slug", p.Slug, "err", err)
                continue
            }
            q := sqlc.New(tx)
            if err := q.SetProfileCodecJS(ctx, sqlc.SetProfileCodecJSParams{ID: p.ID, CodecJs: codecBody}); err != nil {
                _ = tx.Rollback(ctx); continue
            }
            if err := q.MarkProfileSyncedToChirpStack(ctx, sqlc.MarkProfileSyncedToChirpStackParams{ID: p.ID, CsProfileID: csID}); err != nil {
                _ = tx.Rollback(ctx); continue
            }
            if err := tx.Commit(ctx); err != nil { continue }
            synced++
        }
        deps.Log.Info("seed sync complete", "synced", synced, "skipped", len(unsynced)-synced)
    }
