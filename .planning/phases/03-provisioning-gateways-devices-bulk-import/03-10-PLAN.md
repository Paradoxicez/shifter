       Component:
       ```tsx
       export function RevealKeysDialog({ devEUI, open, onOpenChange }: { devEUI: string; open: boolean; onOpenChange: (b: boolean) => void }) {
         const [view, setView] = useState<'idle' | 'success' | 'error'>('idle')
         const [keys, setKeys] = useState<DeviceKeysResponse | null>(null)
         const [errorMsg, setErrorMsg] = useState('')

         const mutation = useMutation({
           mutationFn: () => revealDeviceKeys(devEUI),
           gcTime: 0,  // defense-in-depth — never retain in cache
           onSuccess: (data) => { setKeys(data); setView('success') },
           onError: (err: any) => { setErrorMsg(errorMessageFromApi(err)); setView('error') },
         })

         // Clear state on close
         useEffect(() => {
           if (!open) { setView('idle'); setKeys(null); setErrorMsg('') }
         }, [open])

         return (
           <ResponsiveDialog open={open} onOpenChange={onOpenChange} sm-max-w="md">
             {view === 'idle' && (
               <>
                 <DialogHeader>Reveal device keys</DialogHeader>
                 <p className="text-sm text-muted-foreground">Keys are displayed once. Use Copy to save them — Shifter does not retain a copy.</p>
                 <DialogFooter>
                   <Button variant="ghost" onClick={() => onOpenChange(false)}>Cancel</Button>
                   <Button variant="default" onClick={() => mutation.mutate()}>Reveal keys</Button>
                 </DialogFooter>
               </>
             )}
             {view === 'success' && keys && <KeysPanel keys={keys} onClose={() => onOpenChange(false)} />}
             {view === 'error' && <ErrorState message={errorMsg} onRetry={() => { setView('idle'); mutation.reset() }} />}
           </ResponsiveDialog>
         )
       }
       ```

       Error message map:
       - 403 → "You do not have permission to reveal device keys. Ask an administrator."
       - 404 → "Device not found in Shifter database."
       - 409 → "ChirpStack has no credentials for this device. The device may have been removed externally."
       - 502 → "ChirpStack is unreachable. Try again or check Settings → Connection."
