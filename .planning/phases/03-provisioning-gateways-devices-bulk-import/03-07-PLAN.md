    func addDevice(d Deps) http.HandlerFunc {
        return func(w http.ResponseWriter, r *http.Request) {
            var body struct {
                DevEUI         string  `json:"dev_eui"`
                Name           string  `json:"name"`
                Description    string  `json:"description"`
                ProfileID      uuid.UUID `json:"profile_id"`
                MeteringPointID *uuid.UUID `json:"metering_point_id,omitempty"`
                InitialReading *float64   `json:"initial_reading,omitempty"`
                ActivationMode string  `json:"activation_mode"`  // "OTAA" | "ABP"
                // OTAA fields:
                JoinEUI  string `json:"join_eui,omitempty"`
                AppKey   string `json:"app_key,omitempty"`
                NwkKey   string `json:"nwk_key,omitempty"`  // defaults to AppKey for 1.0.x
                // ABP fields:
                DevAddr  string `json:"dev_addr,omitempty"`
                NwkSKey  string `json:"nwk_s_key,omitempty"`
                AppSKey  string `json:"app_s_key,omitempty"`
                FCntUp   uint32 `json:"fcnt_up,omitempty"`
                FCntDown uint32 `json:"fcnt_down,omitempty"`
            }
            if err := json.NewDecoder(r.Body).Decode(&body); err != nil { /* 400 */ }

            // Validate activation_mode
            switch body.ActivationMode {
            case "OTAA":
                if body.AppKey == "" || body.JoinEUI == "" { /* 400 missing required OTAA fields */ }
                if body.NwkKey == "" { body.NwkKey = body.AppKey }  // 1.0.x default
            case "ABP":
                if body.DevAddr == "" || body.NwkSKey == "" || body.AppSKey == "" { /* 400 */ }
            default:
                /* 400 invalid activation_mode */
            }

            // Phase 2 atomic pattern (open Serializable tx, etc.)
            tx, _ := d.Pool.BeginTx(ctx, pgx.TxOptions{IsoLevel: pgx.Serializable})
            defer tx.Rollback(ctx)
            qtx := q.WithTx(tx)

            // 1. Resolve profile + (optional) MP from Shifter PG.
            // 2. CS CreateDevice (returns CS UUID).
            csUUID, err := d.CS.CreateDevice(ctx, chirpstack.CreateDeviceInput{
                DevEUI: body.DevEUI, Name: body.Name, Description: body.Description,
                JoinEUI: body.JoinEUI, ProfileCSUUID: profile.CsProfileUUID,
            })
            if err != nil { /* 502 */ }

            // 3. Branch on activation_mode
            switch body.ActivationMode {
            case "OTAA":
                if err := d.CS.CreateDeviceWithKeys(ctx, body.DevEUI, body.AppKey, body.NwkKey); err != nil {
                    // Best-effort CS rollback
                    _ = d.CS.DeleteDevice(context.Background(), body.DevEUI)
                    /* 502 */
                }
            case "ABP":
                if err := d.CS.ActivateDevice(ctx, chirpstack.ActivateDeviceInput{
                    DevEUI: body.DevEUI, DevAddr: body.DevAddr,
                    NwkSKey: body.NwkSKey, AppSKey: body.AppSKey,
                    FCntUp: body.FCntUp, FCntDown: body.FCntDown,
                }); err != nil {
                    _ = d.CS.DeleteDevice(context.Background(), body.DevEUI)
                    /* 502 */
                }
            }

            // 4. PG: q.CreateDevice (NO secret columns — DEV-09).
            dev, err := qtx.CreateDevice(ctx, sqlc.CreateDeviceParams{
                DevEui: body.DevEUI, Name: body.Name, DeviceProfileID: body.ProfileID,
                CsDeviceUuid: pgxUUID(csUUID), JoinEui: pgxText(body.JoinEUI), Description: pgxText(body.Description),
            })
            if err != nil {
                _ = d.CS.DeleteDevice(context.Background(), body.DevEUI)
                /* 500 */
            }

            // 5. Optional: open binding if MP supplied (Phase 2 path).
            // 6. Audit row — action='create', source distinguished via Notes if from bulk import.
            audit.WriteEntry(ctx, tx, audit.Entry{
                UserID: actor.ID, Action: audit.ActionCreate,
                EntityType: audit.EntityTypeDevice, EntityID: dev.ID,
                After: map[string]any{"dev_eui": dev.DevEui, "activation_mode": body.ActivationMode},
                RequestID: middleware.GetReqID(ctx),
            })

            if err := tx.Commit(ctx); err != nil {
                _ = d.CS.DeleteDevice(context.Background(), body.DevEUI)
                /* 500 */
            }

            // 7. Echo keys back to client (D-21 success state). Keys are NOT
            // persisted on Shifter side — they're echoed from the request body
            // (operator-supplied) since CS has accepted them.
            resp := map[string]any{
                "id":       dev.ID, "dev_eui": dev.DevEui, "cs_device_uuid": csUUID,
                "activation_mode": body.ActivationMode,
            }
            if body.ActivationMode == "OTAA" {
                resp["join_eui"] = body.JoinEUI
                resp["app_key"]  = body.AppKey
                resp["nwk_key"]  = body.NwkKey
            } else {
                resp["dev_addr"]  = body.DevAddr
                resp["nwk_s_key"] = body.NwkSKey
                resp["app_s_key"] = body.AppSKey
                resp["f_cnt_up"]  = body.FCntUp
                resp["f_cnt_down"]= body.FCntDown
            }
            writeJSON(w, http.StatusCreated, resp)
        }
    }
