    ```go
    package chirpstack

    import (
        "context"
        "sync"
        "time"

        api "github.com/chirpstack/chirpstack/api/go/v4/api"
        "golang.org/x/sync/singleflight"
    )

    // MetricsCache wraps Client.GetMetrics with a 1-minute TTL and
    // singleflight deduplication. Matches CONTEXT D-02 caching strategy
    // (even though the RPC name changed in v4.17 from GetGatewayStats to
    // GetMetrics).

    type cachedMetric struct {
        resp       *api.GetGatewayMetricsResponse
        refreshed  time.Time
    }

    type MetricsCache struct {
        client *Client
        ttl    time.Duration
        mu     sync.RWMutex
        cache  map[string]cachedMetric  // key: gatewayID
        sf     singleflight.Group       // dedupes concurrent fetches
        now    func() time.Time         // injectable for tests
    }

    func NewMetricsCache(client *Client) *MetricsCache {
        return &MetricsCache{
            client: client,
            ttl:    1 * time.Minute,
            cache:  make(map[string]cachedMetric),
            now:    time.Now,
        }
    }

    // Get returns the cached metric if fresh; otherwise fetches via CS GetMetrics
    // (deduplicated via singleflight) and updates the cache.
    func (m *MetricsCache) Get(ctx context.Context, gatewayID string) (*api.GetGatewayMetricsResponse, error) {
        m.mu.RLock()
        c, ok := m.cache[gatewayID]
        m.mu.RUnlock()
        if ok && m.now().Sub(c.refreshed) < m.ttl {
            return c.resp, nil
        }

        // Cache miss / stale — single-flight the fetch.
        v, err, _ := m.sf.Do(gatewayID, func() (interface{}, error) {
            resp, err := m.client.GetMetrics(ctx, GetGatewayMetricsInput{
                GatewayID:   gatewayID,
                Start:       m.now().Add(-24 * time.Hour),
                End:         m.now(),
                Aggregation: "HOUR",
            })
            if err != nil { return nil, err }
            m.mu.Lock()
            m.cache[gatewayID] = cachedMetric{ resp: resp, refreshed: m.now() }
            m.mu.Unlock()
            return resp, nil
        })
        if err != nil { return nil, err }
        return v.(*api.GetGatewayMetricsResponse), nil
    }

    // Invalidate removes a single gateway from the cache (called on
    // decommission / restore to avoid stale display).
    func (m *MetricsCache) Invalidate(gatewayID string) {
        m.mu.Lock()
        delete(m.cache, gatewayID)
        m.mu.Unlock()
    }
    ```
