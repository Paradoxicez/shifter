    ```go
    package importpkg  // package name `importpkg` because `import` is a reserved keyword in Go

    import (
        "encoding/hex"
        "fmt"
        "strings"
        "unicode"
    )

    // normalizeHex is the canonical hex-normalisation helper. Strips spaces,
    // tabs, newlines, hyphens, colons, and the optional 0x prefix; lowercases.
    // Returns the canonical lowercase hex string OR an error describing
    // length / character mismatch.
    func normalizeHex(raw string, wantChars int, label string) (string, error) {
        cleaned := strings.Map(func(r rune) rune {
            switch r {
            case ' ', '-', ':', '\t', '\n', '\r':
                return -1
            }
            return unicode.ToLower(r)
        }, raw)
        cleaned = strings.TrimPrefix(cleaned, "0x")
        if len(cleaned) != wantChars {
            return "", fmt.Errorf("%s: expected %d hex chars after normalisation, got %d", label, wantChars, len(cleaned))
        }
        if _, err := hex.DecodeString(cleaned); err != nil {
            return "", fmt.Errorf("%s: not valid hex: %w", label, err)
        }
        return cleaned, nil
    }

    func NormalizeDevEUI(raw string)   (string, error) { return normalizeHex(raw, 16, "dev_eui") }
    func NormalizeJoinEUI(raw string)  (string, error) { return normalizeHex(raw, 16, "join_eui") }
    func NormalizeAppKey(raw string)   (string, error) { return normalizeHex(raw, 32, "app_key") }
    func NormalizeDevAddr(raw string)  (string, error) { return normalizeHex(raw, 8,  "dev_addr") }
    func NormalizeNwkSKey(raw string)  (string, error) { return normalizeHex(raw, 32, "nwk_s_key") }
    func NormalizeAppSKey(raw string)  (string, error) { return normalizeHex(raw, 32, "app_s_key") }

    // ParseEUI64 is an alias of NormalizeDevEUI for sticker-paste UX (gateway_id
    // and join_eui both follow the EUI64 shape).
    func ParseEUI64(raw string) (string, error) { return NormalizeDevEUI(raw) }
    ```

    ```go
    package importpkg

    import (
        "fmt"
        "io"
        "strings"

        "github.com/xuri/excelize/v2"
    )

    // ParsedRow is the lossless raw + canonicalised form of a single
    // file row. raw_payload preserves every input cell as string (for
    // errors.xlsx round-trip); parsed is populated by the validator.
    type ParsedRow struct {
        RowIndex  int               // 1-based, matches XLSX/CSV row number
        Raw       map[string]string // column name → original cell value
    }

    const sheetName = "Devices"

    // ParseXLSX reads an XLSX upload and returns one ParsedRow per data row.
    // Row 1 must be the header row matching the D-04 column schema.
    // Reject xlsm (macros) by checking the file extension at the handler.
    func ParseXLSX(r io.Reader) ([]ParsedRow, error) {
        f, err := excelize.OpenReader(r)
        if err != nil { return nil, fmt.Errorf("xlsx open: %w", err) }
        defer f.Close()

        // Use the first sheet's name if "Devices" not present (template defaults to it).
        sn := sheetName
        if _, err := f.GetSheetIndex(sn); err != nil || sn == "" {
            sn = f.GetSheetName(0)
        }
        rows, err := f.GetRows(sn)
        if err != nil { return nil, fmt.Errorf("xlsx rows: %w", err) }
        if len(rows) == 0 { return nil, fmt.Errorf("xlsx is empty") }

        header := rows[0]
        out := make([]ParsedRow, 0, len(rows)-1)
        for i, row := range rows[1:] {
            raw := map[string]string{}
            for j, cell := range row {
                if j < len(header) { raw[strings.ToLower(header[j])] = strings.TrimSpace(cell) }
            }
            out = append(out, ParsedRow{ RowIndex: i + 2, Raw: raw })
        }
        return out, nil
    }
    ```

    2. `internal/import/parser_csv.go`:

    ```go
    package importpkg

    import (
        "bufio"
        "bytes"
        "encoding/csv"
        "fmt"
        "io"
        "strings"
        "unicode/utf8"
    )

    const utf8BOM = "\xEF\xBB\xBF"

    // ParseCSV: D-04a UTF-8 only with BOM tolerated. Non-UTF-8 → operator-
    // readable error so the operator re-saves the file rather than getting
    // mojibake silently.
    func ParseCSV(r io.Reader) ([]ParsedRow, error) {
        br := bufio.NewReader(r)
        peek, _ := br.Peek(3)
        if bytes.Equal(peek, []byte(utf8BOM)) { _, _ = br.Discard(3) }

        all, err := io.ReadAll(br)
        if err != nil { return nil, fmt.Errorf("csv read: %w", err) }
        if !utf8.Valid(all) {
            return nil, fmt.Errorf("CSV is not valid UTF-8. Re-save the file as 'CSV UTF-8' or use the XLSX template instead.")
        }
        rdr := csv.NewReader(bytes.NewReader(all))
        rdr.FieldsPerRecord = -1
        rdr.LazyQuotes = true
        rdr.TrimLeadingSpace = true
        rows, err := rdr.ReadAll()
        if err != nil { return nil, fmt.Errorf("csv parse: %w", err) }
        if len(rows) == 0 { return nil, fmt.Errorf("csv is empty") }

        header := rows[0]
        out := make([]ParsedRow, 0, len(rows)-1)
        for i, row := range rows[1:] {
            raw := map[string]string{}
            for j, cell := range row {
                if j < len(header) { raw[strings.ToLower(header[j])] = strings.TrimSpace(cell) }
            }
            out = append(out, ParsedRow{ RowIndex: i + 2, Raw: raw })
        }
        return out, nil
    }
    ```
