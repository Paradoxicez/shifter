**Live-channel banner copy (UI-SPEC §State Conventions):**
- status='reconnecting' → `<Alert variant="warning">` "Reconnecting to live updates…" + sublabel "Last snapshot {N} sec ago" (mono)
- status='reconnecting' AND (Date.now() - lastEventAt) > 30_000 → "Live updates paused" + "Showing last-known values from {time}"
- status='closed' (401) → "Session expired" + link to /login
- status='open' → render nothing (no green banner)

**Live KPI recomputation from SSE deltas (D-06):**
- Maintain a per-MP latest-instant-value map in component state
- On each measurement event for an MP, update the map's entry for that MP
- Re-derive `instant_total` per utility by summing the map's entries for MPs of that utility (the snapshot's `latest_readings` provides the utility_class lookup)
- The other 3 KPIs (today, delta, online) recompute via the SSE-triggered `invalidateQueries(['dashboard','snapshot'])` in useSSE — the snapshot refetches and the tiles re-render with fresh values. No client-side projection.

```tsx
import { useEffect, useMemo, useState } from 'react'
import { useQuery } from '@tanstack/react-query'
import { useDashboardScope } from '@/hooks/useDashboardScope'
import { useSSE, type MeasurementDelta } from '@/hooks/useSSE'
import { apiFetch } from '@/lib/apiFetch'
import { KpiGrid } from '@/components/dashboard/KpiGrid'
import { LiveChannelBanner } from '@/components/dashboard/LiveChannelBanner'
import { EmptyStateOnboarding } from '@/components/dashboard/EmptyStateOnboarding'
import { Skeleton } from '@/components/ui/skeleton'

interface Snapshot { /* shape per Plan 04 */ }

export default function DashboardPage() {
  const scope = useDashboardScope()
  const snapshot = useQuery<Snapshot>({
    queryKey: ['dashboard', 'snapshot'],
    queryFn: () => apiFetch<Snapshot>('/api/dashboard/snapshot'),
    enabled: scope.data?.onboarding.uplink_count != null && scope.data.onboarding.uplink_count > 0,
  })

  // Live instant map: per-MP latest instant_value
  const [latestInstantMap, setLatestInstantMap] = useState<Record<string, { utility: 'water' | 'electricity'; instant: number | null }>>({})

  // Seed map from snapshot once it lands
  useEffect(() => {
    if (snapshot.data) {
      const seed: typeof latestInstantMap = {}
      for (const r of snapshot.data.latest_readings) {
        seed[r.metering_point_id] = { utility: r.utility_class as 'water' | 'electricity', instant: r.instant_value }
      }
      setLatestInstantMap(seed)
    }
  }, [snapshot.data])

  const sse = useSSE({
    topics: ['dashboard:global'],
    onMeasurement: (delta: MeasurementDelta) => {
      setLatestInstantMap(prev => {
        const existing = prev[delta.metering_point_id]
        if (!existing) return prev   // unknown MP — will appear on next snapshot
        return { ...prev, [delta.metering_point_id]: { ...existing, instant: delta.instant_value } }
      })
    },
  })

  const instantOverride = useMemo(() => {
    const sums = { water: 0, electricity: 0 }
    for (const entry of Object.values(latestInstantMap)) {
      if (entry.instant != null) sums[entry.utility] += entry.instant
    }
    return sums
  }, [latestInstantMap])

  if (scope.isLoading) return <Skeleton className="h-96" />
  if (!scope.data) return null

  const { capabilities, onboarding } = scope.data

  // D-21 empty state
  if (onboarding.uplink_count === 0) {
    return (
      <div className="space-y-8">
        <LiveChannelBanner status={sse.status} lastEventAt={sse.lastEventAt} />
        <EmptyStateOnboarding onboarding={onboarding} />
      </div>
    )
  }

  return (
    <div className="space-y-8">
      <LiveChannelBanner status={sse.status} lastEventAt={sse.lastEventAt} />
      <div className="flex items-baseline justify-between">
        <h1 className="text-2xl font-semibold leading-8">Dashboard</h1>
        {/* Plan 08 inserts DateRangePicker here */}
      </div>
      {snapshot.data && (
        <KpiGrid
          capabilities={capabilities}
          kpis={snapshot.data.kpis}
          instantOverride={instantOverride}
        />
      )}
      {/* Plan 08 inserts ConsumptionChart cards here */}
    </div>
  )
}
```
