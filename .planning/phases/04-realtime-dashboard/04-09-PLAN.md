1. Create `JsonTree.tsx` (use the pattern from RESEARCH.md §Pattern 7):

```tsx
import { useState } from 'react'
import { ChevronDown, ChevronRight } from 'lucide-react'

interface JsonTreeProps {
  value: unknown
  name?: string
  depth?: number
  defaultOpen?: boolean
}

const MAX_DEPTH = 32

export function JsonTree({ value, name, depth = 0, defaultOpen = false }: JsonTreeProps) {
  const [open, setOpen] = useState(defaultOpen)
  if (depth > MAX_DEPTH) return <span className="text-muted-foreground italic">…truncated</span>
  if (value === null) return <Row><Key>{name}</Key>: <span className="text-muted-foreground italic font-mono text-sm">null</span></Row>
  if (typeof value === 'string') return <Row><Key>{name}</Key>: <span className="text-success font-mono text-sm">{`"${value}"`}</span></Row>
  if (typeof value === 'number') return <Row><Key>{name}</Key>: <span className="text-info font-mono text-sm">{String(value)}</span></Row>
  if (typeof value === 'boolean') return <Row><Key>{name}</Key>: <span className="text-warning font-mono text-sm">{String(value)}</span></Row>

  const isArray = Array.isArray(value)
  const entries = isArray ? (value as unknown[]).map((v, i) => [String(i), v] as [string, unknown]) : Object.entries(value as Record<string, unknown>)

  return (
    <div>
      <button
        type="button"
        aria-expanded={open}
        onClick={() => setOpen(o => !o)}
        className="flex items-center gap-1 text-sm font-mono font-semibold"
      >
        {open ? <ChevronDown className="h-3 w-3" /> : <ChevronRight className="h-3 w-3" />}
        <Key>{name ?? '(root)'}</Key>
        <span className="text-xs text-muted-foreground">{`(${entries.length} item${entries.length === 1 ? '' : 's'})`}</span>
      </button>
      {open && (
        <div style={{ paddingLeft: 16 * (depth + 1) }}>
          {entries.map(([k, v]) => (
            <JsonTree key={k} name={k} value={v} depth={depth + 1} defaultOpen={k === 'extra' && depth === 0} />
          ))}
        </div>
      )}
    </div>
  )
}

function Row({ children }: { children: React.ReactNode }) {
  return <div className="text-sm font-mono">{children}</div>
}
function Key({ children }: { children: React.ReactNode }) {
  return <span className="text-foreground font-semibold">{children}</span>
}
```

```tsx
import { useParams, useSearchParams } from 'react-router-dom'
import { useQuery } from '@tanstack/react-query'
import { useState } from 'react'
import { Tabs, TabsList, TabsTrigger, TabsContent } from '@/components/ui/tabs'
import { Badge } from '@/components/ui/badge'
import { Tooltip, TooltipContent, TooltipTrigger } from '@/components/ui/tooltip'
import { apiFetch } from '@/lib/apiFetch'
import { useSSE, type MeasurementDelta } from '@/hooks/useSSE'
import { NormalTab } from '@/components/metering-point/NormalTab'
import { AdvancedTab } from '@/components/metering-point/AdvancedTab'
import { UplinksLogTab } from '@/components/metering-point/UplinksLogTab'

export default function MeteringPointDetailPage() {
  const { id } = useParams<{ id: string }>()
  const [params, setParams] = useSearchParams()
  const activeTab = params.get('tab') ?? 'normal'

  const detailQuery = useQuery({
    queryKey: ['mp', id, 'detail'],
    queryFn: () => apiFetch(`/api/metering-points/${id}`),
    enabled: !!id,
  })
  const signalQuery = useQuery({
    queryKey: ['mp', id, 'signal-history'],
    queryFn: () => apiFetch(`/api/metering-points/${id}/signal-history`),
    enabled: !!id && detailQuery.data?.latest_reading != null,
  })

  const [pendingPayload, setPendingPayload] = useState<any | null>(null)
  useSSE({
    topics: [`mp:${id}`, `mp:${id}:uplinks`],
    onMeasurement: (delta: MeasurementDelta) => {
      if (delta.metering_point_id === id) {
        // Defer auto-swap of Advanced tab; queue pending payload
        // We rely on detail-query refetch (useSSE invalidates ['mp', id, 'detail'] on snapshot)
        // For Advanced tab we set the pending flag here so it can show the alert
        setPendingPayload(delta)
      }
    },
  })

  if (!detailQuery.data) return null
  const { metering_point: mp, latest_reading } = detailQuery.data
  const tabsDisabled = latest_reading == null

  return (
    <div className="space-y-6">
      <div>
        <h1 className="text-2xl font-semibold leading-8">{mp.name}</h1>
        <div className="text-sm text-muted-foreground">
          {mp.site_name ?? '—'} · <Badge variant="secondary">{mp.utility_class}</Badge>
        </div>
      </div>
      <Tabs value={activeTab} onValueChange={(v) => setParams(p => { p.set('tab', v); return p })}>
        <TabsList>
          <TabsTrigger value="normal">Normal</TabsTrigger>
          {tabsDisabled ? (
            <Tooltip><TooltipTrigger asChild><span><TabsTrigger value="advanced" disabled>Advanced</TabsTrigger></span></TooltipTrigger>
              <TooltipContent>Available after first uplink</TooltipContent></Tooltip>
          ) : <TabsTrigger value="advanced">Advanced</TabsTrigger>}
          {tabsDisabled ? (
            <Tooltip><TooltipTrigger asChild><span><TabsTrigger value="uplinks" disabled>Uplinks log</TabsTrigger></span></TooltipTrigger>
              <TooltipContent>Available after first uplink</TooltipContent></Tooltip>
          ) : <TabsTrigger value="uplinks">Uplinks log</TabsTrigger>}
        </TabsList>
        <TabsContent value="normal" className="mb-2">
          <NormalTab detail={detailQuery.data} signalHistory={signalQuery.data} online={detailQuery.data.online} />
        </TabsContent>
        <TabsContent value="advanced" className="mb-2">
          {latest_reading && <AdvancedTab latestReading={latest_reading} pendingPayload={pendingPayload} clearPending={() => setPendingPayload(null)} />}
        </TabsContent>
        <TabsContent value="uplinks" className="mb-2">
          {latest_reading && <UplinksLogTab meteringPointId={id!} />}
        </TabsContent>
      </Tabs>
    </div>
  )
}
```
