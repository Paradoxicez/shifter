**SSE wire format (verbatim — match exactly):**

```
HTTP/1.1 200 OK
Content-Type: text/event-stream
Cache-Control: no-cache
Connection: keep-alive
X-Accel-Buffering: no

event: snapshot
data: {"ready":true,"topics":["dashboard:global","mp:abc-123"]}

: heartbeat

event: measurement
data: {"metering_point_id":"abc-123","time":"2026-05-11T12:34:56.789Z","cumulative_value":1234.56,"instant_value":4.2,"quality":"ok","battery_pct":87,"rssi":-65}

: heartbeat
```

Each frame ends with `\n\n`. Heartbeat is a comment line `: heartbeat\n\n` every 30s. SSE comments are ignored by the EventSource API but keep the TCP connection alive.

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-03-01 | Denial of service | unauthenticated SSE drains process FDs | mitigate | Route mounted under authenticated group (SCS LoadAndSave + session-required middleware). Anonymous request → 401 (existing middleware behavior). |

| T-04-03-03 | Information disclosure | viewer subscribes to another user's MP topic | accept | All MPs belong to the single install; D-23 explicitly grants viewers full read. No multi-tenant isolation in scope. |
| T-04-03-04 | Tampering | malformed topic string causes regex DoS | mitigate | Topic regex anchored + bounded length (max 64 chars per topic, max 64 topics per connection). Reject 400 on overflow. |
| T-04-03-05 | Repudiation | which user received which event | accept | SSE delivery is not audited (Phase 4 introduces no mutations; audit is for state changes). |
</threat_model>

1. Create `internal/events/handler.go` with:

```go
package events

import (
    "context"
    "encoding/json"
    "errors"
    "fmt"
    "net/http"
    "regexp"
    "strings"
    "time"

    "github.com/google/uuid"
)

const (
    maxTopicsPerConnection = 64
    heartbeatInterval      = 30 * time.Second
)

var topicRegex = regexp.MustCompile(`^(dashboard:global|mp:[0-9a-f-]{36}(:uplinks)?)$`)

type Deps struct {
    Hub    *Hub
    Logger *slog.Logger
}

func (d Deps) handleSSE(w http.ResponseWriter, r *http.Request) {
    topics, err := parseTopics(r.URL.Query().Get("topics"))
    if err != nil {
        http.Error(w, err.Error(), http.StatusBadRequest)
        return
    }

    flusher, ok := w.(http.Flusher)
    if !ok {
        http.Error(w, "streaming unsupported", http.StatusInternalServerError)
        return
    }

    w.Header().Set("Content-Type", "text/event-stream")
    w.Header().Set("Cache-Control", "no-cache")
    w.Header().Set("Connection", "keep-alive")
    w.Header().Set("X-Accel-Buffering", "no")
    w.WriteHeader(http.StatusOK)

    connID := uuid.NewString()
    ch := d.Hub.Subscribe(connID, topics)
    defer d.Hub.Unsubscribe(connID)

    // Snapshot-ready marker per D-03. The actual snapshot payload is fetched
    // by the client from REST endpoints (Plan 04 dashboard/snapshot, Plan 05
    // metering-points/:id/detail). The marker tells the client "you're live;
    // go fetch your snapshot now."
    snapshotMsg, _ := json.Marshal(map[string]any{"ready": true, "topics": topics})
    fmt.Fprintf(w, "event: snapshot\ndata: %s\n\n", snapshotMsg)
    flusher.Flush()

    heartbeat := time.NewTicker(heartbeatInterval)
    defer heartbeat.Stop()

    for {
        select {
        case <-r.Context().Done():
            return
        case payload, open := <-ch:
            if !open {
                return
            }
            fmt.Fprintf(w, "event: measurement\ndata: %s\n\n", payload)
            flusher.Flush()
        case <-heartbeat.C:
            fmt.Fprint(w, ": heartbeat\n\n")
            flusher.Flush()
        }
    }
}

func parseTopics(raw string) ([]string, error) {
    if raw == "" {
        return nil, errors.New("topics query param required")
    }
    parts := strings.Split(raw, ",")
    if len(parts) > maxTopicsPerConnection {
        return nil, fmt.Errorf("too many topics (max %d)", maxTopicsPerConnection)
    }
    for _, t := range parts {
        if !topicRegex.MatchString(t) {
            return nil, fmt.Errorf("invalid topic: %q", t)
        }
    }
    return parts, nil
}
```
