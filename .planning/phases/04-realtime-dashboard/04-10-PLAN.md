     await page.goto('/login')

   export async function injectMeasurement(opts: {
     meteringPointId: string
     cumulativeValue: number
     instantValue: number
     quality?: string
   }) {
     // Use the Phase 2 testharness CLI subcommand if available.
     // Discover via: grep -r "test-harness" cmd/shifter/ — if a subcommand exists,
     // invoke it with the required args. Otherwise, use psql directly via DATABASE_URL.

   }

   export async function setInstallCapabilities(value: 'water' | 'electricity' | 'both') {
     // Approach A (preferred): test-harness subcommand
