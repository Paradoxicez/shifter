**Snapshot event handling (D-03):**
On every `snapshot` event:
1. Reset `attempt` to 0
2. Set status to `'open'`
3. Invalidate `['dashboard','snapshot']` query key (Plan 07's dashboard refetches)
4. For each `mp:<uuid>` topic the hook is subscribed to, invalidate `['mp', uuid, 'detail']`

Snapshot event from Plan 03 emits `{ ready: true, topics: ["dashboard:global", "mp:abc"] }` — exactly two keys (`ready`, `topics`). There is NO `conn_id` in the wire payload (Plan 03 ships no subscribe/unsubscribe API, so no client-side correlation id is needed). Extract `topics` and derive the MP UUIDs.

**TypeScript snapshot type:**

```typescript
interface SnapshotEvent {
  ready: boolean
  topics: string[]
  // NOTE: no conn_id field — Phase 4 ships no subscribe/unsubscribe HTTP API.
}
```

**Measurement event handling:**
1. Update `lastEventAt = Date.now()`
2. Call `opts.onMeasurement(delta)` if provided
3. Apply default invalidation:
   - `['mp', delta.metering_point_id, 'latest']` → setQueryData with the delta
   - `['dashboard','snapshot']` → invalidate (KPI tiles recompute)
   - If on dashboard timeseries page with range ∈ {today, 24h} → invalidate `['dashboard','timeseries', range]` (Plan 07 owns this — pass via `invalidationKeys` override)

**One EventSource per tab (D-01) + topic-set changes:**
- The hook MAY be called in multiple components but they share a single underlying EventSource via a module-level registry
- Topic set is the UNION of all callers' topics
- When the union changes (caller mounts/unmounts/changes topics), the hook **closes the EventSource and reconnects with a new `?topics=...` query string**. There is no `POST /api/events/subscribe` endpoint to call (Plan 03 ships SSE only — see Plan 03 <design_notes>).
- This matches D-04 reconnect-with-snapshot semantics — every topic-set change triggers a fresh snapshot anyway, so the React-Query cache invalidation path is uniform whether the reconnect was from network loss or topic change.
- Document this rationale in a top-of-file comment in `useSSE.ts`.

**Auth-expired heuristic:** EventSource doesn't expose HTTP status. After a hard close, fire a probe `fetch('/api/account/me', { credentials: 'include' })`. If 401, set status to `'closed'` and don't retry; surface to caller via status. Otherwise resume backoff.

```typescript
import { useEffect, useRef, useState, useCallback } from 'react'
import { useQueryClient } from '@tanstack/react-query'

// Phase 4 topic-set changes are handled by close + reconnect-with-new-?topics=.
// We do NOT call any /api/events/subscribe HTTP endpoint — Plan 03 ships SSE only
// (see Plan 03 <design_notes>). Every reconnect emits a fresh snapshot via D-03,
// which keeps the React-Query invalidation path uniform across network loss and
// topic-set changes.

export type SSEStatus = 'connecting' | 'open' | 'reconnecting' | 'closed'

export interface MeasurementDelta {
  metering_point_id: string
  time: string
  cumulative_value: number | null
  instant_value: number | null
  quality: 'ok' | 'decode_fail' | 'missing_canonical' | 'out_of_range' | 'duplicate_fcnt'
  battery_pct: number | null
  rssi: number | null
}

export interface SnapshotEvent {
  ready: boolean
  topics: string[]
  // NOTE: no conn_id — Phase 4 ships no subscribe/unsubscribe HTTP API.
}

export interface UseSSEOptions {
  topics: string[]
  onMeasurement?: (delta: MeasurementDelta) => void
  invalidationKeys?: (delta: MeasurementDelta) => Array<string | (string | number)[]>
}

export interface UseSSEResult {
  status: SSEStatus
  attempt: number
  lastEventAt: number | null
}

export function useSSE(opts: UseSSEOptions): UseSSEResult {
  const queryClient = useQueryClient()
  const [status, setStatus] = useState<SSEStatus>('connecting')
  const [attempt, setAttempt] = useState(0)
  const [lastEventAt, setLastEventAt] = useState<number | null>(null)
  const esRef = useRef<EventSource | null>(null)

  // Topics serialized for dependency stability — sort so order doesn't trigger reconnects.
  const topicsKey = opts.topics.slice().sort().join(',')

  const connect = useCallback(() => {
    if (topicsKey === '') return
    const url = `/api/events?topics=${encodeURIComponent(topicsKey)}`
    const es = new EventSource(url, { withCredentials: true })
    esRef.current = es
    setStatus('connecting')

    es.addEventListener('open', () => {
      setAttempt(0)
      setStatus('open')
    })

    es.addEventListener('snapshot', (ev) => {
      try {
        const snap = JSON.parse((ev as MessageEvent).data) as SnapshotEvent
        // D-03: always-full-snapshot — invalidate dashboard snapshot key on every snapshot event
        queryClient.invalidateQueries({ queryKey: ['dashboard', 'snapshot'] })
        // For each mp:<uuid> topic, invalidate the per-MP detail key
        for (const t of snap.topics ?? []) {
          if (t.startsWith('mp:')) {
            const id = t.slice(3).split(':')[0] // strip optional ':uplinks' suffix
            queryClient.invalidateQueries({ queryKey: ['mp', id, 'detail'] })
          }
        }
        setLastEventAt(Date.now())
      } catch (err) {
        console.warn('useSSE: failed to parse snapshot event', err)
      }
    })

    es.addEventListener('measurement', (ev) => {
      try {
        const delta = JSON.parse((ev as MessageEvent).data) as MeasurementDelta
        queryClient.setQueryData(['mp', delta.metering_point_id, 'latest'], delta)
        queryClient.invalidateQueries({ queryKey: ['dashboard', 'snapshot'] })
        if (opts.invalidationKeys) {
          for (const k of opts.invalidationKeys(delta)) {
            queryClient.invalidateQueries({ queryKey: Array.isArray(k) ? k : [k] })
          }
        }
        opts.onMeasurement?.(delta)
        setLastEventAt(Date.now())
      } catch (err) {
        console.warn('useSSE: failed to parse measurement event', err)
      }
    })

    es.addEventListener('error', async () => {
      es.close()
      esRef.current = null
      setStatus('reconnecting')

      // 401 probe — EventSource can't surface HTTP status
      try {
        const probe = await fetch('/api/account/me', { credentials: 'include' })
        if (probe.status === 401) {
          setStatus('closed')
          return
        }
      } catch {
        // ignore network error on probe; treat as transient
      }

      // D-04 backoff: min(30s, 0.5s * 2^n) + jitter
      setAttempt((prev) => {
        const next = prev + 1
        const delay = Math.min(30_000, 500 * 2 ** next) + Math.random() * 1000
        window.setTimeout(connect, delay)
        return next
      })
    })
  }, [topicsKey, queryClient, opts.onMeasurement, opts.invalidationKeys])

  useEffect(() => {
    connect()
    return () => {
      esRef.current?.close()
      esRef.current = null
      setStatus('closed')
    }
  }, [connect])

  return { status, attempt, lastEventAt }
}
```
